13    /// Analyse `data` (raw EVTX bytes) and return all detected anomalies,
14    /// sorted from highest severity to lowest.
15    pub fn analyse(data: &[u8]) -> Vec<IntegrityAnomaly> {
16        let mut out: Vec<IntegrityAnomaly> = Vec::new();
17
18        // ── Minimum container size guard ──────────────────────────────────────
19        // A valid EVTX file header is exactly 128 bytes. Anything shorter cannot
20        // contain a parseable header and must be treated as a truncated file.
21        if data.len() < 128 {
22            out.push(IntegrityAnomaly::TruncatedFile {
23                declared_chunks: 0,
24                found_chunks: 0,
25            });
26            return out;
27        }
28
29        // ── File header checks ────────────────────────────────────────────────
30        let file_header = if data[0..8] == ELFFILE_MAGIC {
31            out.extend(verify_file_header_checksum(&data[0..0x80]));
32            let fh = EvtxFileHeader::parse(&data[0..128]);
33            if let Some(ref fh) = fh {
34                out.extend(check_file_flags(fh.file_flags));
35                // Emit EmptyLog when the header declares zero chunks.
36                if fh.chunk_count == 0 {
37                    out.push(IntegrityAnomaly::EmptyLog);
38                }
39            }
40            fh
41        } else {
42            None
43        };
44
45        // ── Walk chunks ───────────────────────────────────────────────────────
46        let chunk_size = CHUNK_SIZE as usize;
47        // Start after file header if present, otherwise scan from beginning.
48        let start = if file_header.is_some() { 4096 } else { 0 };
49
50        let mut chunks: Vec<(u64, EvtxChunkHeader)> = Vec::new(); // (offset, header)
51        // (record_id, chunk_offset, header_timestamp) for export-corruption detection
52        let mut all_records_export: Vec<(u64, u64, u64)> = Vec::new();
53        // (record_id, timestamp_ns as i64) for phantom detection
54        let mut all_records_phantom: Vec<(u64, i64)> = Vec::new();
55        // GUIDs per chunk for GUID consistency
56        let mut guids: Vec<[u8; 16]> = Vec::new();
57
58        let mut i = start;
59        while i + 8 <= data.len() {
60            if data[i..i + 8] != ELFCHNK_MAGIC {
61                i += 8;
62                continue;
63            }
64
65            let chunk_end = i + chunk_size;
66            if chunk_end > data.len() {
67                // Truncated chunk — still validate what we can
68                if let Some(hdr) = EvtxChunkHeader::parse(&data[i..]) {
69                    out.extend(check_chunk_data_length(hdr.free_space_offset));
70                }
71                break;
72            }
73
74            let chunk_data = &data[i..chunk_end];
75            let chunk_offset = i as u64;
76
77            if let Some(hdr) = EvtxChunkHeader::parse(chunk_data) {
78                // Chunk header checksum
79                out.extend(verify_chunk_header_checksum(chunk_data, chunk_offset));
80
81                // Records area checksum
82                out.extend(verify_records_area_checksum(chunk_data, chunk_offset));
83
84                // Data length sanity
85                out.extend(check_chunk_data_length(hdr.free_space_offset));
86
87                // DanderSpritz deletion
88                out.extend(detect_danderspritz_deletion(chunk_data, chunk_offset));
89
90                // Extract GUID (bytes 96..112 in chunk header per EVTX spec)
91                if chunk_data.len() >= 112 {
92                    let mut guid = [0u8; 16];
93                    guid.copy_from_slice(&chunk_data[96..112]);
94                    guids.push(guid);
95                }
96
97                // Walk records for timestamp / phantom / export checks
98                let records_start = CHUNK_RECORDS_OFFSET as usize;
99                let free_end = hdr.free_space_offset as usize;
100                let records_end = free_end.min(chunk_size);
101
102                let mut rec_pairs: Vec<(u64, u64)> = Vec::new(); // (record_id, ts) for monotonicity
103                let mut pos = records_start;
104                while pos + 4 <= records_end.min(chunk_data.len()) {
105                    if chunk_data[pos..pos + 4] != RECORD_MAGIC {
106                        break;
107                    }
108                    if let Some(rhdr) = EvtxRecordHeader::parse(&chunk_data[pos..]) {
109                        let sz = rhdr.size as usize;
110                        if sz < 28 || pos + sz > records_end.min(chunk_data.len()) {
111                            break;
112                        }
113                        rec_pairs.push((rhdr.record_id, rhdr.timestamp));
114                        all_records_export.push((rhdr.record_id, chunk_offset, rhdr.timestamp));
115                        all_records_phantom
116                            .push((rhdr.record_id, rhdr.timestamp as i64));
117                        pos += sz;
118                    } else {
119                        break;
120                    }
121                }
122
123                // Timestamp monotonicity per chunk
124                out.extend(check_timestamp_monotonicity(&rec_pairs, chunk_offset));
125
126                chunks.push((chunk_offset, hdr));
127            }
128
129            i += chunk_size;
130        }
131
132        // ── Cross-chunk checks ────────────────────────────────────────────────
133
134        // Record ID gaps across chunks
135        let chunk_ranges: Vec<(u64, u64)> = chunks
136            .iter()
137            .map(|(_, h)| (h.first_event_record_number, h.last_event_record_number))
138            .collect();
139        out.extend(detect_record_id_gaps(&chunk_ranges));
140
141        // File header consistency (next_record_id vs actual highest)
142        if let Some(ref fh) = file_header {
143            let actual_highest = chunks
144                .iter()
145                .map(|(_, h)| h.last_event_record_id)
146                .max()
147                .unwrap_or(0);
148            out.extend(check_file_header_consistency(
149                fh.next_record_id,
150                actual_highest,
151            ));
152            out.extend(check_chunk_count(fh.chunk_count, chunks.len()));
153        }
154
155        // GUID consistency across chunks
156        out.extend(check_log_file_guid_consistency(&guids));
157
158        // Export timestamp corruption (Fox-IT 2019)
159        out.extend(detect_export_timestamp_corruption(&all_records_export));
160
161        // Phantom record injection — convert suspicious PhantomAlerts to IntegrityAnomaly
162        all_records_phantom.sort_by_key(|&(id, _)| id);
163        out.extend(phantom_alerts_to_anomalies(&detect_phantom_records(&all_records_phantom)));
164
165        // ── Dedup + sort by severity descending ───────────────────────────────
166        // Use Debug repr for dedup (IntegrityAnomaly derives Debug).
167        let mut seen = std::collections::HashSet::new();
168        out.retain(|a| seen.insert(format!("{a:?}")));
169
170        out.sort_by_key(|a| std::cmp::Reverse(a.severity()));
171        out
172    }
173
174    /// Same as `analyse` but also accepts a severity floor — only anomalies
175    /// at or above `min_severity` are returned.
176    pub fn analyse_min_severity(data: &[u8], min_severity: Severity) -> Vec<IntegrityAnomaly> {
177        Self::analyse(data)
178            .into_iter()
179            .filter(|a| a.severity() >= min_severity)
180            .collect()
181    }
182}

187/// Given (first_record_number, last_record_number) per chunk in order,
188/// detect gaps between adjacent chunks.
189pub fn detect_record_id_gaps(chunks: &[(u64, u64)]) -> Vec<IntegrityAnomaly> {
190    let mut out = Vec::new();
191    for window in chunks.windows(2) {
192        let (_, prev_last) = window[0];
193        let (next_first, _) = window[1];
194        let expected = prev_last + 1;
195        if next_first != expected {
196            out.push(IntegrityAnomaly::RecordIdGap {
197                chunk_offset: 0, // caller fills in real offset
198                expected,
199                found: next_first,
200            });
201        }
202    }
203    out
204}

206/// Verify the chunk header CRC32 (bytes 0x00..0x78) against stored value at 0x78.
207/// `buf` must be at least 0x7C bytes. `chunk_offset` is for the indicator.
208pub fn verify_chunk_header_checksum(buf: &[u8], chunk_offset: u64) -> Vec<IntegrityAnomaly> {
209    if buf.len() < 0x7C {
210        return vec![];
211    }
212    let stored = u32::from_le_bytes(buf[0x78..0x7C].try_into().unwrap_or([0; 4]));
213    let computed = crc32fast::hash(&buf[0..0x78]);
214    if stored != computed {
215        vec![IntegrityAnomaly::ChunkChecksumMismatch {
216            chunk_offset,
217            computed,
218            stored,
219        }]
220    } else {
221        vec![]
222    }
223}

225/// Check that (record_id, timestamp) pairs are monotonically non-decreasing in timestamp.
226pub fn check_timestamp_monotonicity(
227    records: &[(u64, u64)],
228    chunk_offset: u64,
229) -> Vec<IntegrityAnomaly> {
230    let mut out = Vec::new();
231    let mut prev_ts = 0u64;
232    for &(record_id, ts) in records {
233        if ts < prev_ts {
234            out.push(IntegrityAnomaly::TimestampAnomaly {
235                chunk_offset,
236                record_id,
237                prev_ts,
238                this_ts: ts,
239            });
240        }
241        prev_ts = ts;
242    }
243    out
244}

246/// Verify the file header CRC32 (bytes 0x00..0x78) against stored value at 0x7C.
247/// `buf` must be at least 0x80 bytes (128 bytes).
248pub fn verify_file_header_checksum(buf: &[u8]) -> Vec<IntegrityAnomaly> {
249    if buf.len() < 0x80 {
250        return vec![];
251    }
252    let stored = u32::from_le_bytes(buf[0x7C..0x80].try_into().unwrap_or([0; 4]));
253    let computed = crc32fast::hash(&buf[0..0x78]);
254    if stored != computed {
255        vec![IntegrityAnomaly::FileHeaderChecksumMismatch { computed, stored }]
256    } else {
257        vec![]
258    }
259}

261/// Check file flags for dirty/full anomalies.
262/// Bit 0x1 = not cleanly shut down; bit 0x2 = file full.
263pub fn check_file_flags(flags: u32) -> Vec<IntegrityAnomaly> {
264    let mut out = Vec::new();
265    if flags & 0x1 != 0 {
266        out.push(IntegrityAnomaly::FileNotCleanlyShutdown);
267    }
268    if flags & 0x2 != 0 {
269        out.push(IntegrityAnomaly::FileFull);
270    }
271    out
272}

286/// Verify the event records area CRC32 for a chunk.
287/// Records area = bytes `0x200..free_space_offset` (from chunk header field at bytes 48..52).
288/// CRC32 compared with `event_records_checksum` at bytes 52..56.
289pub fn verify_records_area_checksum(chunk_data: &[u8], chunk_offset: u64) -> Vec<IntegrityAnomaly> {
290    if chunk_data.len() < 0x38 {
291        return vec![];
292    }
293    let free_space_offset =
294        u32::from_le_bytes(chunk_data[48..52].try_into().unwrap_or([0; 4])) as usize;
295    let stored = u32::from_le_bytes(chunk_data[52..56].try_into().unwrap_or([0; 4]));
296    let records_start = 0x200usize;
297    if free_space_offset < records_start || chunk_data.len() < free_space_offset {
298        return vec![];
299    }
300    let computed = crc32fast::hash(&chunk_data[records_start..free_space_offset]);
301    if stored != computed {
302        vec![IntegrityAnomaly::RecordChecksumMismatch {
303            chunk_offset,
304            computed,
305            stored,
306        }]
307    } else {
308        vec![]
309    }
310}

312/// Check file header consistency: next_record_id should be > actual_highest_record_id.
313pub fn check_file_header_consistency(
314    header_next: u64,
315    actual_highest: u64,
316) -> Vec<IntegrityAnomaly> {
317    if header_next <= actual_highest {
318        vec![IntegrityAnomaly::NextRecordIdInconsistency {
319            header_next,
320            actual_highest,
321        }]
322    } else {
323        vec![]
324    }

327/// Validate that a chunk's data-length field is within the legal EVTX range [512, 65536].
328///
329/// Chunk data lengths outside this range indicate a structurally corrupt or hand-crafted chunk.
330pub fn check_chunk_data_length(length: u32) -> Vec<IntegrityAnomaly> {
331    if !(512..=65536).contains(&length) {
332        vec![IntegrityAnomaly::InvalidChunkDataLength(length)]

338/// Check that the `log_file_guid` is identical across all chunks.
339///
340/// `guids` is a slice of 16-byte GUIDs, one per chunk in file order.  A mismatch at any index
341/// implies the chunk was transplanted from a different log file, which is an anti-forensic
342/// indicator.  Returns one `LogFileGuidMismatch` per mismatching chunk (compared to chunk 0).
343pub fn check_log_file_guid_consistency(guids: &[[u8; 16]]) -> Vec<IntegrityAnomaly> {
344    if guids.len() <= 1 {
345        return vec![];
346    }
347    let first = guids[0];
348    guids.iter().enumerate().skip(1)
349        .filter(|(_, g)| **g != first)
350        .map(|(i, g)| IntegrityAnomaly::LogFileGuidMismatch {
351            chunk_index: i,
352            expected: u128::from_le_bytes(first),
353            actual: u128::from_le_bytes(*g),
354        })
355        .collect()
356}

358/// A suspicious gap in the record ID sequence that may indicate phantom record injection.
359#[derive(Debug, Clone, serde::Serialize)]
360pub struct PhantomAlert {
361    /// First record ID in the gap (inclusive).
362    pub gap_start_id: u64,
363    /// Last record ID in the gap (inclusive).
364    pub gap_end_id: u64,
365    /// Timestamp (nanoseconds) of the record immediately before the gap.
366    pub prev_timestamp_ns: i64,
367    /// Timestamp (nanoseconds) of the record immediately after the gap.
368    pub next_timestamp_ns: i64,
369    /// True when the timestamp gap does not proportionally explain the record ID gap,
370    /// suggesting records were injected without advancing the clock.
371    pub suspicious: bool,
372}
373
374/// Detect phantom record injection by correlating record-ID gaps with timestamp gaps.
375///
376/// `records` is a slice of `(record_id, timestamp_ns)` pairs sorted by `record_id`.
377///
378/// A record-ID gap where the elapsed nanoseconds per missing record is less than 1 ms
379/// (1_000_000 ns) is flagged as suspicious — real log activity cannot normally produce
380/// millions of events per second.  A large timestamp jump alongside the gap (≥ 1 ms per
381/// missing ID) is treated as a normal cleared-log or ring-buffer rollover period.
382pub fn detect_phantom_records(records: &[(u64, i64)]) -> Vec<PhantomAlert> {
383    // 1 ms per missing record: below this rate the gap is suspicious.
384    const MIN_NS_PER_MISSING_RECORD: i64 = 1_000_000;
385
386    let mut out = Vec::new();
387    for window in records.windows(2) {
388        let (prev_id, prev_ts) = window[0];
389        let (next_id, next_ts) = window[1];
390        if next_id != prev_id + 1 {
391            let gap_count = (next_id - prev_id - 1) as i64;
392            let ts_gap = next_ts - prev_ts;
393            let suspicious = ts_gap <= 0 || ts_gap / gap_count < MIN_NS_PER_MISSING_RECORD;
394            out.push(PhantomAlert {
395                gap_start_id: prev_id + 1,
396                gap_end_id: next_id - 1,
397                prev_timestamp_ns: prev_ts,
398                next_timestamp_ns: next_ts,
399                suspicious,
400            });
401        }
402    }
403    out
404}

406/// Convert `PhantomAlert` results from [`detect_phantom_records`] into
407/// [`IntegrityAnomaly::PhantomRecordInjection`] entries.
408///
409/// Only alerts where `suspicious == true` are converted; benign gaps (large
410/// timestamp delta proportional to missing record count) are silently dropped.
411pub fn phantom_alerts_to_anomalies(alerts: &[PhantomAlert]) -> Vec<IntegrityAnomaly> {
412    alerts
413        .iter()
414        .filter(|a| a.suspicious)
415        .map(|a| IntegrityAnomaly::PhantomRecordInjection {
416            gap_start_id: a.gap_start_id,
417            gap_end_id: a.gap_end_id,
418            prev_timestamp_ns: a.prev_timestamp_ns,
419            next_timestamp_ns: a.next_timestamp_ns,
420        })
421        .collect()
422}

424/// Detect the wevtutil / Event Viewer export timestamp corruption described by Fox-IT.
425///
426/// When `wevtutil epl` or "Save As…" exports an EVTX, each record's header timestamp is
427/// replaced with the **previous** record's BinXml timestamp.  The first record in the export
428/// therefore receives no predecessor and gets timestamp = 0.
429///
430/// This function returns one [`IntegrityAnomaly::ExportTimestampCorruption`] for every
431/// record whose header timestamp is 0.
432///
433/// `records` is a slice of `(record_id, chunk_offset, header_timestamp)` tuples.
434///
435/// Reference: Wassenaar, Fox-IT BV (2019).
436/// "Export corrupts Windows Event Log files"
437/// <https://blog.fox-it.com/2019/06/04/export-corrupts-windows-event-log-files/>
438pub fn detect_export_timestamp_corruption(records: &[(u64, u64, u64)]) -> Vec<IntegrityAnomaly> {
439    records
440        .iter()
441        .filter(|&&(_, _, ts)| ts == 0)
442        .map(
443            |&(record_id, chunk_offset, _)| IntegrityAnomaly::ExportTimestampCorruption {
444                record_id,
445                chunk_offset,
446            },
447        )
448        .collect()
449}

451/// Detect surgical record deletion consistent with the NSA DanderSpritz `eventlogedit` tool.
452///
453/// The tool absorbs a deleted record into the preceding record's size field: the absorbing
454/// record's `size` is inflated to span the bytes of the deleted record, whose magic bytes
455/// (`0x2A 0x2A 0x00 0x00`) remain visible inside the inflated body.  No EID 1102 is emitted.
456///
457/// This function walks the records area (`0x200..free_space_offset`) of `chunk_data` and
458/// emits one [`IntegrityAnomaly::SurgicalRecordDeletion`] for each record whose stated
459/// body contains the record-magic pattern.
460///
461/// `chunk_offset` is the byte offset of the chunk within the source file (used in indicators).
462///
463/// Reference: Wassenaar & van Dijk, Fox-IT BV (2017).
464/// "Detection and recovery of NSA's covered up tracks"
465/// <https://blog.fox-it.com/2017/12/08/detection-and-recovery-of-nsas-covered-up-tracks/>
466///
467/// Reference implementation (Python, MIT License): fox-it/danderspritz-evtx
468/// <https://github.com/fox-it/danderspritz-evtx>
469/// Algorithm independently re-implemented in Rust.
470pub fn detect_danderspritz_deletion(chunk_data: &[u8], chunk_offset: u64) -> Vec<IntegrityAnomaly> {
471    const RECORD_MAGIC: [u8; 4] = [0x2A, 0x2A, 0x00, 0x00];
472    const RECORDS_START: usize = 0x200;
473    const FREE_SPACE_OFF_FIELD: usize = 0x30; // bytes 48..52 in chunk header
474
475    if chunk_data.len() < FREE_SPACE_OFF_FIELD + 4 {
476        return vec![];
477    }
478    let free_space_offset = u32::from_le_bytes(
479        chunk_data[FREE_SPACE_OFF_FIELD..FREE_SPACE_OFF_FIELD + 4]
480            .try_into()
481            .unwrap_or([0; 4]),
482    ) as usize;
483
484    if free_space_offset <= RECORDS_START || chunk_data.len() < free_space_offset {
485        return vec![];
486    }
487
488    let mut indicators = Vec::new();
489    let mut pos = RECORDS_START;
490
491    while pos + 24 <= free_space_offset {
492        // Every valid record starts with magic.
493        if chunk_data[pos..pos + 4] != RECORD_MAGIC {
494            break;
495        }
496
497        let stated_size =
498            u32::from_le_bytes(chunk_data[pos + 4..pos + 8].try_into().unwrap_or([0; 4])) as usize;
499
500        // Minimum well-formed record: 4 magic + 4 size + 8 id + 8 ts + 0 payload + 4 tail = 28
501        if stated_size < 28 || pos + stated_size > free_space_offset {
502            break;
503        }
504
505        let record_id =
506            u64::from_le_bytes(chunk_data[pos + 8..pos + 16].try_into().unwrap_or([0; 8]));
507
508        // Body = bytes after the 24-byte header, before the 4-byte trailing size field.
509        let body_start = pos + 24;
510        let body_end = pos + stated_size - 4;
511
512        // Search the body for a nested record-magic pattern.
513        if body_end > body_start {
514            let body = &chunk_data[body_start..body_end];
515            if let Some(rel) = body.windows(4).position(|w| w == RECORD_MAGIC) {
516                let ghost_offset_in_chunk = (body_start + rel) as u64;
517                indicators.push(IntegrityAnomaly::SurgicalRecordDeletion {
518                    chunk_offset,
519                    absorbing_record_id: record_id,
520                    stated_size: stated_size as u32,
521                    ghost_offset_in_chunk,
522                });
523            }
524        }
525
526        pos += stated_size;
527    }
528
529    indicators
530}

810        let mut chunk = vec![0u8; 0x10000];
811        chunk[0..8].copy_from_slice(b"ElfChnk\0");
812        chunk[8..16].copy_from_slice(&1u64.to_le_bytes());
813        chunk[16..24].copy_from_slice(&1u64.to_le_bytes());
814        chunk[24..32].copy_from_slice(&1u64.to_le_bytes());
815        chunk[32..40].copy_from_slice(&1u64.to_le_bytes());
816        chunk[40..44].copy_from_slice(&0x80u32.to_le_bytes());
817        chunk[44..48].copy_from_slice(&0x200u32.to_le_bytes());
818        // free_space_offset = 0x200 (empty records area)
819        chunk[48..52].copy_from_slice(&0x200u32.to_le_bytes());
820        // CRC32 of empty slice (0x200..0x200) = 0x00000000
821        chunk[52..56].copy_from_slice(&0u32.to_le_bytes());
822        let crc = crc32fast::hash(&chunk[0..0x78]);
823        chunk[0x78..0x7C].copy_from_slice(&crc.to_le_bytes());
824        chunk
825    }

926    // ── detect_danderspritz_deletion ─────────────────────────────────────────
927    //
928    // Reference: Wassenaar & van Dijk, Fox-IT BV (2017).
929    // "Detection and recovery of NSA's covered up tracks"
930    // https://blog.fox-it.com/2017/12/08/detection-and-recovery-of-nsas-covered-up-tracks/
931    //
932    // Reference implementation (Python, MIT License):
933    // fox-it/danderspritz-evtx — https://github.com/fox-it/danderspritz-evtx
934    // Algorithm independently re-implemented in Rust.
935
936    fn write_record_at(
937        buf: &mut Vec<u8>,
938        offset: usize,
939        record_id: u64,
940        ts: u64,
941        payload: &[u8],
942    ) -> u32 {
943        let size = (4 + 4 + 8 + 8 + payload.len() + 4) as u32;
944        let end = offset + size as usize;
945        if buf.len() < end {
946            buf.resize(end, 0);
947        }
948        buf[offset..offset + 4].copy_from_slice(&[0x2A, 0x2A, 0x00, 0x00]);
949        buf[offset + 4..offset + 8].copy_from_slice(&size.to_le_bytes());
950        buf[offset + 8..offset + 16].copy_from_slice(&record_id.to_le_bytes());
951        buf[offset + 16..offset + 24].copy_from_slice(&ts.to_le_bytes());
952        buf[offset + 24..offset + 24 + payload.len()].copy_from_slice(payload);
953        let tail = offset + size as usize - 4;
954        buf[tail..tail + 4].copy_from_slice(&size.to_le_bytes());
955        size
956    }
