1/// Anomaly detected when an event's provider metadata is inconsistent with its content.
2#[derive(Debug, Clone, PartialEq, Eq, serde::Serialize)]
3pub enum ProviderAnomaly {
4    /// The event ID is not in the expected set for this provider GUID.
5    UnexpectedEventId {
6        provider_name: String,
7        event_id: u32,
8    },
9    /// The provider name matches a known provider, but the GUID does not.
10    GuidSpoofing {
11        provider_name: String,
12        expected_guid: [u8; 16],
13        actual_guid: [u8; 16],
14    },
15    /// The channel does not match the expected channel for this provider.
16    ChannelMismatch {
17        provider_name: String,
18        expected_channel: String,
19        actual_channel: String,
20    },
21}

93/// Check provider metadata consistency for a single event.
94///
95/// `provider_guid` is the 16-byte GUID as it appears in the BinXML data.
96/// If `None`, only the name-based checks are performed.
97///
98/// Checks:
99/// 1. If `provider_name` is in the known table: verify GUID matches.
100/// 2. If GUID matches a known table entry: verify `event_id` is expected.
101/// 3. If GUID matches a known table entry: verify `channel` matches.
102pub fn check_provider_consistency(
103    provider_name: &str,
104    provider_guid: Option<&[u8; 16]>,
105    event_id: u32,
106    channel: &str,
107) -> Vec<ProviderAnomaly> {
108    let mut out = Vec::new();
109
110    // Find entry by name
111    let by_name = KNOWN_PROVIDERS.iter().find(|p| {
112        p.name.eq_ignore_ascii_case(provider_name)
113    });
114
115    // Find entry by GUID
116    let by_guid = provider_guid.and_then(|g| {
117        KNOWN_PROVIDERS.iter().find(|p| &p.guid == g)
118    });
119
120    // 1. Provider name matches known entry — check GUID
121    if let (Some(entry), Some(guid)) = (by_name, provider_guid) {
122        if &entry.guid != guid {
123            out.push(ProviderAnomaly::GuidSpoofing {
124                provider_name: provider_name.to_string(),
125                expected_guid: entry.guid,
126                actual_guid: *guid,
127            });
128        }
129    }
130
131    // 2 & 3. GUID matches known entry — check event_id and channel
132    if let Some(entry) = by_guid {
133        if !entry.event_ids.contains(&event_id) {
134            out.push(ProviderAnomaly::UnexpectedEventId {
135                provider_name: entry.name.to_string(),
136                event_id,
137            });
138        }
139        if !entry.channel.eq_ignore_ascii_case(channel) {
140            out.push(ProviderAnomaly::ChannelMismatch {
141                provider_name: entry.name.to_string(),
142                expected_channel: entry.channel.to_string(),
143                actual_channel: channel.to_string(),
144            });
145        }
146    }
147
148    out
149}

152impl forensicnomicon::report::Observation for ProviderAnomaly {
153    fn severity(&self) -> Option<forensicnomicon::report::Severity> {
154        use forensicnomicon::report::Severity;
155        Some(match self {
156            ProviderAnomaly::GuidSpoofing { .. } => Severity::High,
157            ProviderAnomaly::ChannelMismatch { .. } => Severity::Medium,
158            ProviderAnomaly::UnexpectedEventId { .. } => Severity::Low,
159        })
160    }
161
162    fn category(&self) -> forensicnomicon::report::Category {
163        use forensicnomicon::report::Category;
164        match self {
165            ProviderAnomaly::GuidSpoofing { .. } => Category::Concealment,
166            ProviderAnomaly::ChannelMismatch { .. } => Category::Structure,
167            ProviderAnomaly::UnexpectedEventId { .. } => Category::Provenance,
168        }
169    }
170
171    fn code(&self) -> &'static str {
172        match self {
173            ProviderAnomaly::UnexpectedEventId { .. } => "WINEVT-PROVIDER-UNEXPECTED-EVENTID",
174            ProviderAnomaly::GuidSpoofing { .. } => "WINEVT-PROVIDER-GUID-SPOOFING",
175            ProviderAnomaly::ChannelMismatch { .. } => "WINEVT-PROVIDER-CHANNEL-MISMATCH",
176        }
177    }
178
179    fn note(&self) -> String {
180        match self {
181            ProviderAnomaly::UnexpectedEventId { provider_name, event_id } => {
182                format!("event ID {event_id} is not in the expected set for provider '{provider_name}'")
183            }
184            ProviderAnomaly::GuidSpoofing { provider_name, .. } => {
185                format!("provider name '{provider_name}' matches a known provider but its GUID does not")
186            }
187            ProviderAnomaly::ChannelMismatch { provider_name, expected_channel, actual_channel } => {
188                format!("provider '{provider_name}' logged to '{actual_channel}', expected '{expected_channel}'")
189            }
190        }
191    }
192
193    fn mitre(&self) -> &'static [&'static str] {
194        match self {
195            ProviderAnomaly::GuidSpoofing { .. } => &["T1036"],
196            _ => &[],
197        }
198    }
199
200    fn evidence(&self) -> Vec<forensicnomicon::report::Evidence> {
201        let ev = |field: &str, value: String| forensicnomicon::report::Evidence {
202            field: field.to_string(),
203            value,
204            location: None,
205        };
206        let hex = |g: &[u8; 16]| g.iter().map(|b| format!("{b:02x}")).collect::<String>();
207        match self {
208            ProviderAnomaly::UnexpectedEventId { provider_name, event_id } => vec![
209                ev("provider_name", provider_name.clone()),
210                ev("event_id", event_id.to_string()),
211            ],
212            ProviderAnomaly::GuidSpoofing { provider_name, expected_guid, actual_guid } => vec![
213                ev("provider_name", provider_name.clone()),
214                ev("expected_guid", hex(expected_guid)),
215                ev("actual_guid", hex(actual_guid)),
216            ],
217            ProviderAnomaly::ChannelMismatch { provider_name, expected_channel, actual_channel } => vec![
218                ev("provider_name", provider_name.clone()),
219                ev("expected_channel", expected_channel.clone()),
220                ev("actual_channel", actual_channel.clone()),
221            ],
222        }
223    }
224}
