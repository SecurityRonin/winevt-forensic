1//! Structured field analysis of Windows Event Log records.
2//!
3//! Builds on the `evtx` crate (full BinXml parser) to extract per-event
4//! fields — event ID, logon session LUIDs, PowerShell script block text,
5//! event frequency distributions — from intact or reconstructed EVTX files.
6//!
7//! For corrupt or cleared EVTX files, first reconstruct with
8//! `winevt_carver::carve_from_file` + `winevt_writer::records_to_evtx`,
9//! then pass the reconstructed path here.
10

113/// Extract the integer EventID from the `Event.System.EventID` field.
114///
115/// The `evtx` crate may represent this as:
116/// - `4624` (bare integer)
117/// - `{"#text": 4624, "#attributes": {"Qualifiers": 0}}` (with Qualifiers)
118fn event_id_from_system(system: &serde_json::Value) -> Option<u32> {
119    let raw = system.get("EventID")?;
120    if let Some(n) = raw.as_u64() {
121        return Some(n as u32);
122    }
123    // Object form: { "#text": N, ... }
124    raw.get("#text")
125        .and_then(serde_json::Value::as_u64)
126        .map(|n| n as u32)
127}

129/// Extract a string field from `EventData` by name.
130/// Read a string field from EventData, handling both EVTX serialization shapes:
131///
132/// 1. Named-attribute format (Security log, most audit events):
133///    `{"Data": [{"@Name": "key", "#text": "value"}, ...]}`
134/// 2. Named-element format (Sysmon EID 1, PowerShell EID 4104, etc.):
135///    `{"Image": "value", "ScriptBlockText": "..."}`
136///
137/// The two shapes come from how the provider defines its `<template>` in the
138/// manifest: `<Data Name="…">` serializes as shape 1; bare `<Image>` elements
139/// serialize as shape 2.
140fn event_data_str<'a>(event_data: &'a serde_json::Value, key: &str) -> Option<&'a str> {
141    // Shape 1: named-attribute array
142    if let Some(arr) = event_data.get("Data").and_then(|d| d.as_array()) {
143        if let Some(hit) = arr.iter().find_map(|item| {
144            if item.get("@Name")?.as_str()? == key {
145                item.get("#text").and_then(|v| v.as_str())
146            } else {
147                None
148            }
149        }) {
150            return Some(hit);
151        }
152    }
153    // Shape 2: flat named-element object (Sysmon, PowerShell, WMI-Activity…)
154    event_data.get(key)?.as_str()
155}

157/// Read a string field from Sysmon EventData (flat object format).
158fn sysmon_str<'a>(event_data: &'a serde_json::Value, key: &str) -> Option<&'a str> {
159    event_data.get(key)?.as_str()
160}

169/// Flatten a decoded EVTX record's `EventData` and `UserData` payload into a
170/// flat `field name → value` map.
171///
172/// `event_record` is one record's JSON as produced by the `evtx` crate's
173/// `records_json_value()` (the `{"Event": {...}}` object). The two
174/// provider-manifest serialization shapes are normalized into the same flat
175/// map:
176///
177/// 1. **Named-attribute array** (Security / audit events, `<Data Name="…">`):
178///    `{"Data": [{"@Name": "TargetUserName", "#text": "jdoe"}, …]}`
179/// 2. **Flat named-element object** (Sysmon, PowerShell, WMI, bare `<Image>`):
180///    `{"Image": "C:\\…", "CommandLine": "…"}`
181///
182/// Extraction is **lossless**: every leaf value is preserved (no truncation,
183/// no field cap), scalars are stringified, and unnamed `<Data>` elements are
184/// retained under positional keys so nothing is silently dropped. `UserData`
185/// (provider-specific, often nested) is flattened recursively. Returns an empty
186/// map when neither block is present.
187#[must_use]
188pub fn flatten_event_data(event_record: &serde_json::Value) -> BTreeMap<String, String> {
189    let mut out = BTreeMap::new();
190    for block in ["EventData", "UserData"] {
191        if let Some(node) = event_record.get("Event").and_then(|e| e.get(block)) {
192            collect_fields(node, &mut out);
193        }
194    }
195    out
196}
197
198/// Recursively collect leaf `name → value` pairs from an `EventData`/`UserData`
199/// node, normalizing the two EVTX serialization shapes. Panic-free.
200fn collect_fields(node: &serde_json::Value, out: &mut BTreeMap<String, String>) {
201    let serde_json::Value::Object(map) = node else {
202        return;
203    };
204    for (key, value) in map {
205        // Skip JSON-meta attribute keys at object level (@Name, @Guid, xmlns…).
206        if key.starts_with('@') || key == "#text" {
207            continue;
208        }
209        if key == "Data" {
210            collect_data_field(value, out);
211            continue;
212        }
213        match value {
214            // Nested named element (UserData provider blocks).
215            serde_json::Value::Object(_) => collect_fields(value, out),
216            serde_json::Value::Array(arr) => {
217                for (i, item) in arr.iter().enumerate() {
218                    if item.is_object() {
219                        collect_fields(item, out);
220                    } else if let Some(s) = scalar_to_string(item) {
221                        out.insert(format!("{key}{i}"), s);
222                    }
223                }
224            }
225            scalar => {
226                if let Some(s) = scalar_to_string(scalar) {
227                    out.insert(key.clone(), s);
228                }
229            }
230        }
231    }
232}

234/// Handle the named-attribute `Data` array shape and its single-element variant.
235fn collect_data_field(value: &serde_json::Value, out: &mut BTreeMap<String, String>) {
236    match value {
237        serde_json::Value::Array(items) => {
238            for (i, item) in items.iter().enumerate() {
239                insert_data_item(i, item, out);
240            }
241        }
242        other => insert_data_item(0, other, out),
243    }
244}
245
246/// Insert one `<Data>` element: `{"@Name": k, "#text": v}`, a nested object, or
247/// an unnamed scalar kept under a positional `Data{index}` key.
248fn insert_data_item(index: usize, item: &serde_json::Value, out: &mut BTreeMap<String, String>) {
249    if let serde_json::Value::Object(obj) = item {
250        if let Some(name) = obj.get("@Name").and_then(serde_json::Value::as_str) {
251            let val = obj.get("#text").and_then(scalar_to_string).unwrap_or_default();
252            out.insert(name.to_string(), val);
253        } else {
254            collect_fields(item, out);
255        }
256    } else if let Some(s) = scalar_to_string(item) {
257        out.insert(format!("Data{index}"), s);
258    }
259}

261/// Stringify a JSON scalar; returns `None` for null or composite values.
262fn scalar_to_string(v: &serde_json::Value) -> Option<String> {
263    match v {
264        serde_json::Value::String(s) => Some(s.clone()),
265        serde_json::Value::Number(n) => Some(n.to_string()),
266        serde_json::Value::Bool(b) => Some(b.to_string()),
267        _ => None,
268    }
269}

271/// Parse an EVTX file and return all events sorted by timestamp.
272///
273/// Records with unparseable timestamps are included with an empty string
274/// timestamp and sorted to the end.
275pub fn timeline(path: &Path) -> Result<Vec<TimelineEntry>, AnalyzeError> {
276    // Verify path is readable before handing to the evtx crate (gives a
277    // friendlier error via our AnalyzeError::Io variant).
278    let _ = std::fs::metadata(path).map_err(AnalyzeError::Io)?;
279
280    let mut parser =
281        evtx::EvtxParser::from_path(path).map_err(|e| AnalyzeError::Parse(e.to_string()))?;
282
283    let mut entries: Vec<TimelineEntry> = Vec::new();
284    for result in parser.records_json_value() {
285        let record = match result {
286            Ok(r) => r,
287            Err(_) => continue, // skip unparseable records
288        };
289        let system = record.data.get("Event").and_then(|e| e.get("System"));
290
291        let event_id = system.and_then(event_id_from_system).unwrap_or(0);
292
293        let level = system
294            .and_then(|s| s.get("Level"))
295            .and_then(serde_json::Value::as_u64)
296            .map(|n| n as u8);
297
298        let channel = system
299            .and_then(|s| s.get("Channel"))
300            .and_then(serde_json::Value::as_str)
301            .map(str::to_owned);
302
303        let computer = system
304            .and_then(|s| s.get("Computer"))
305            .and_then(serde_json::Value::as_str)
306            .map(str::to_owned);
307
308        let provider = system
309            .and_then(|s| s.get("Provider"))
310            .and_then(|p| {
311                // Provider can be {"@Name": "...", "@Guid": "..."}
312                p.get("@Name").or_else(|| p.get("@Guid"))
313            })
314            .and_then(serde_json::Value::as_str)
315            .map(str::to_owned);
316
317        entries.push(TimelineEntry {
318            record_id: record.event_record_id,
319            timestamp: record.timestamp.to_string(),
320            event_id,
321            level,
322            channel,
323            computer,
324            provider,
325        });
326    }
327
328    // Sort by timestamp string (ISO-8601 sorts lexicographically)
329    entries.sort_by(|a, b| a.timestamp.cmp(&b.timestamp));
330    Ok(entries)
331}

333/// Reconstruct logon sessions from EID 4624 / 4634 / 4647 events.
334///
335/// Sessions are keyed on the `TargetLogonId` LUID field.  Logoff events
336/// (EID 4634 and 4647) are matched to the most recent logon with the
337/// same `TargetLogonId`.  Sessions with no matching logoff have
338/// `logoff_time = None`.
339pub fn sessions(path: &Path) -> Result<Vec<LogonSession>, AnalyzeError> {
340    let _ = std::fs::metadata(path).map_err(AnalyzeError::Io)?;
341
342    let mut parser =
343        evtx::EvtxParser::from_path(path).map_err(|e| AnalyzeError::Parse(e.to_string()))?;
344
345    // logon_id → LogonSession (open sessions)
346    let mut open: HashMap<String, LogonSession> = HashMap::new();
347    // Completed sessions (logoff matched)
348    let mut closed: Vec<LogonSession> = Vec::new();
349    // Preserve insertion order for open sessions
350    let mut insertion_order: Vec<String> = Vec::new();
351
352    for result in parser.records_json_value() {
353        let record = match result {
354            Ok(r) => r,
355            Err(_) => continue,
356        };
357        let event = &record.data;
358        let system = match event.get("Event").and_then(|e| e.get("System")) {
359            Some(s) => s,
360            None => continue,
361        };
362        let event_id = match event_id_from_system(system) {
363            Some(id) => id,
364            None => continue,
365        };
366        let ts = record.timestamp.to_string();
367        let event_data = event.get("Event").and_then(|e| e.get("EventData"));
368
369        match event_id {
370            // EID 4624 — An account was successfully logged on
371            4624 => {
372                let ed = match event_data {
373                    Some(d) => d,
374                    None => continue,
375                };
376                let logon_id = event_data_str(ed, "TargetLogonId")
377                    .unwrap_or("-")
378                    .to_owned();
379                let username = event_data_str(ed, "TargetUserName")
380                    .unwrap_or("-")
381                    .to_owned();
382                let domain = event_data_str(ed, "TargetDomainName")
383                    .unwrap_or("-")
384                    .to_owned();
385                let logon_type = event_data_str(ed, "LogonType")
386                    .and_then(|s| s.parse().ok())
387                    .unwrap_or(0);
388                let ip_raw = event_data_str(ed, "IpAddress").map(str::to_owned);
389                let ip_address = ip_raw.filter(|ip| ip != "-" && !ip.is_empty());
390
391                let session = LogonSession {
392                    logon_id: logon_id.clone(),
393                    username,
394                    domain,
395                    logon_type,
396                    ip_address,
397                    logon_time: Some(ts),
398                    logoff_time: None,
399                    duration_secs: None,
400                };
401                if !open.contains_key(&logon_id) {
402                    insertion_order.push(logon_id.clone());
403                }
404                open.insert(logon_id, session);
405            }
406            // EID 4634 — An account was logged off
407            // EID 4647 — User initiated logoff
408            4634 | 4647 => {
409                let ed = match event_data {
410                    Some(d) => d,
411                    None => continue,
412                };
413                let logon_id = match event_data_str(ed, "TargetLogonId") {
414                    Some(id) => id.to_owned(),
415                    None => continue,
416                };
417                if let Some(mut session) = open.remove(&logon_id) {
418                    session.logoff_time = Some(ts.clone());
419                    // Compute duration using jiff::Timestamp arithmetic
420                    if let (Some(logon), Some(logoff)) = (
421                        session.logon_time.as_deref(),
422                        session.logoff_time.as_deref(),
423                    ) {
424                        if let (Ok(t0), Ok(t1)) = (
425                            logon.parse::<jiff::Timestamp>(),
426                            logoff.parse::<jiff::Timestamp>(),
427                        ) {
428                            let d = t1.duration_since(t0).as_secs();
429                            session.duration_secs = Some(d);
430                        }
431                    }
432                    closed.push(session);
433                    insertion_order.retain(|id| id != &logon_id);
434                }
435            }
436            _ => {}
437        }
438    }
439
440    // Append still-open sessions in insertion order
441    let mut result = closed;
442    for id in &insertion_order {
443        if let Some(s) = open.remove(id) {
444            result.push(s);
445        }
446    }
447    Ok(result)
448}

450/// Correlate logon sessions across **multiple** EVTX files.
451///
452/// Unlike [`sessions`] (which is file-scoped), this function merges raw
453/// 4624/4634/4647 events from all supplied paths, sorts them by timestamp,
454/// and then runs the logon/logoff correlation state machine once — so a
455/// session whose logon lands in one file and whose logoff lands in another
456/// is correctly paired and its duration computed.
457///
458/// Returns `Err` if **every** path is unreadable.  Paths that cannot be
459/// parsed are silently skipped so a corrupt EVTX in a directory does not
460/// abort the entire scan.
461///
462/// Passing an empty slice returns `Ok(vec![])`.
463pub fn sessions_multi(paths: &[&Path]) -> Result<Vec<LogonSession>, AnalyzeError> {
464    if paths.is_empty() {
465        return Ok(Vec::new());
466    }
467
468    // Collect (timestamp_str, event_id, event_data_json) triples from all files.
469    struct RawEvent {
470        timestamp: String,
471        event_id: u32,
472        data: serde_json::Value,
473    }
474
475    let mut all_raw: Vec<RawEvent> = Vec::new();
476    let mut any_ok = false;
477
478    for &path in paths {
479        match evtx::EvtxParser::from_path(path) {
480            Err(_) => continue,
481            Ok(mut parser) => {
482                any_ok = true;
483                for result in parser.records_json_value() {
484                    let record = match result { Ok(r) => r, Err(_) => continue };
485                    let system = record.data.get("Event").and_then(|e| e.get("System"));
486                    let event_id = match system.and_then(event_id_from_system) {
487                        Some(id) if matches!(id, 4624 | 4634 | 4647) => id,
488                        _ => continue,
489                    };
490                    all_raw.push(RawEvent {
491                        timestamp: record.timestamp.to_string(),
492                        event_id,
493                        data: record.data,
494                    });
495                }
496            }
497        }
498    }
499
500    if !any_ok {
501        return Err(AnalyzeError::Io(std::io::Error::new(
502            std::io::ErrorKind::NotFound,
503            "none of the supplied paths could be opened",
504        )));
505    }
506
507    // Sort by timestamp so the correlation state machine sees events in order.
508    all_raw.sort_by(|a, b| a.timestamp.cmp(&b.timestamp));
509
510    // Run the same correlation logic as sessions().
511    let mut open: HashMap<String, LogonSession> = HashMap::new();
512    let mut closed: Vec<LogonSession> = Vec::new();
513    let mut insertion_order: Vec<String> = Vec::new();
514
515    for raw in &all_raw {
516        let ed = match raw.data.get("Event").and_then(|e| e.get("EventData")) {
517            Some(d) => d,
518            None => continue,
519        };
520        match raw.event_id {
521            4624 => {
522                let logon_id = event_data_str(ed, "TargetLogonId").unwrap_or("-").to_owned();
523                let username  = event_data_str(ed, "TargetUserName").unwrap_or("-").to_owned();
524                let domain    = event_data_str(ed, "TargetDomainName").unwrap_or("-").to_owned();
525                let logon_type = event_data_str(ed, "LogonType")
526                    .and_then(|s| s.parse().ok()).unwrap_or(0);
527                let ip_address = event_data_str(ed, "IpAddress")
528                    .map(str::to_owned)
529                    .filter(|ip| ip != "-" && !ip.is_empty());
530                let session = LogonSession {
531                    logon_id: logon_id.clone(), username, domain, logon_type,
532                    ip_address, logon_time: Some(raw.timestamp.clone()),
533                    logoff_time: None, duration_secs: None,
534                };
535                if !open.contains_key(&logon_id) {
536                    insertion_order.push(logon_id.clone());
537                }
538                open.insert(logon_id, session);
539            }
540            4634 | 4647 => {
541                let logon_id = match event_data_str(ed, "TargetLogonId") {
542                    Some(id) => id.to_owned(),
543                    None => continue,
544                };
545                if let Some(mut session) = open.remove(&logon_id) {
546                    session.logoff_time = Some(raw.timestamp.clone());
547                    if let (Some(logon), Some(logoff)) =
548                        (session.logon_time.as_deref(), session.logoff_time.as_deref())
549                    {
550                        if let (Ok(t0), Ok(t1)) = (
551                            logon.parse::<jiff::Timestamp>(),
552                            logoff.parse::<jiff::Timestamp>(),
553                        ) {
554                            session.duration_secs = Some(t1.duration_since(t0).as_secs());
555                        }
556                    }
557                    closed.push(session);
558                    insertion_order.retain(|id| id != &logon_id);
559                }
560            }
561            _ => {}
562        }
563    }
564
565    let mut result = closed;
566    for id in &insertion_order {
567        if let Some(s) = open.remove(id) {
568            result.push(s);
569        }
570    }
571    Ok(result)
572}

574/// Build a merged logon graph from **multiple** EVTX files.
575///
576/// Calls [`logon_graph`] on each readable path and merges the results:
577/// nodes are unioned, edges with the same `(source, target, logon_type)`
578/// tuple have their counts summed.  Paths that fail to parse are skipped.
579///
580/// Passing an empty slice returns an empty graph (not an error).
581pub fn logon_graph_multi(paths: &[&Path]) -> Result<LogonGraph, AnalyzeError> {
582    let mut node_set = std::collections::HashSet::new();
583    let mut edge_map: HashMap<(String, String, u32), usize> = HashMap::new();
584
585    for &path in paths {
586        if let Ok(g) = logon_graph(path) {
587            for node in g.nodes { node_set.insert(node); }
588            for edge in g.edges {
589                *edge_map.entry((edge.source, edge.target, edge.logon_type))
590                    .or_insert(0) += edge.count;
591            }
592        }
593    }
594
595    let mut nodes: Vec<String> = node_set.into_iter().collect();
596    nodes.sort();
597    let mut edges: Vec<LogonEdge> = edge_map.into_iter()
598        .map(|((source, target, logon_type), count)| LogonEdge { source, target, logon_type, count })
599        .collect();
600    edges.sort_by(|a, b| a.source.cmp(&b.source).then(a.target.cmp(&b.target)));
601
602    Ok(LogonGraph { nodes, edges })
603}

605/// Reassemble PowerShell script blocks from EID 4104 events.
606///
607/// Groups events by `ScriptBlockId`, sorts fragments by `MessageNumber`,
608/// and concatenates `ScriptBlockText` values.  Returns one `ScriptBlock`
609/// per unique GUID, in the order the first fragment was observed.
610pub fn powershell_blocks(path: &Path) -> Result<Vec<ScriptBlock>, AnalyzeError> {
611    let _ = std::fs::metadata(path).map_err(AnalyzeError::Io)?;
612
613    let mut parser =
614        evtx::EvtxParser::from_path(path).map_err(|e| AnalyzeError::Parse(e.to_string()))?;
615
616    // script_block_id → (path, Vec<(message_number, text)>)
617    type BlockEntry = (Option<String>, Vec<(u32, String)>);
618    let mut blocks: HashMap<String, BlockEntry> = HashMap::new();
619    let mut insertion_order: Vec<String> = Vec::new();
620
621    for result in parser.records_json_value() {
622        let record = match result {
623            Ok(r) => r,
624            Err(_) => continue,
625        };
626        let event = &record.data;
627        let system = match event.get("Event").and_then(|e| e.get("System")) {
628            Some(s) => s,
629            None => continue,
630        };
631        if event_id_from_system(system) != Some(4104) {
632            continue;
633        }
634        let event_data = match event.get("Event").and_then(|e| e.get("EventData")) {
635            Some(d) => d,
636            None => continue,
637        };
638
639        let script_id = match event_data_str(event_data, "ScriptBlockId") {
640            Some(id) => id.to_owned(),
641            None => continue,
642        };
643        let text = event_data_str(event_data, "ScriptBlockText")
644            .unwrap_or("")
645            .to_owned();
646        let msg_num: u32 = event_data_str(event_data, "MessageNumber")
647            .and_then(|s| s.parse().ok())
648            .unwrap_or(1);
649        let script_path = event_data_str(event_data, "Path")
650            .filter(|p| !p.is_empty())
651            .map(str::to_owned);
652
653        let entry = blocks.entry(script_id.clone()).or_insert_with(|| {
654            insertion_order.push(script_id.clone());
655            (script_path, Vec::new())
656        });
657        entry.1.push((msg_num, text));
658    }
659
660    let mut result = Vec::with_capacity(blocks.len());
661    for id in insertion_order {
662        if let Some((path_val, mut parts)) = blocks.remove(&id) {
663            parts.sort_by_key(|(n, _)| *n);
664            let count = parts.len() as u32;
665            let text = parts
666                .into_iter()
667                .map(|(_, t)| t)
668                .collect::<Vec<_>>()
669                .join("");
670            result.push(ScriptBlock {
671                script_block_id: id,
672                text,
673                path: path_val,
674                parts: count,
675            });
676        }
677    }
678    Ok(result)
679}

681/// Compute a frequency distribution of event IDs.
682///
683/// Useful for spotting bursts of a single event ID that may indicate
684/// brute-force attacks, log flooding, or other anomalies.
685pub fn frequency(path: &Path) -> Result<FrequencyReport, AnalyzeError> {
686    let _ = std::fs::metadata(path).map_err(AnalyzeError::Io)?;
687
688    let mut parser =
689        evtx::EvtxParser::from_path(path).map_err(|e| AnalyzeError::Parse(e.to_string()))?;
690
691    let mut counts: HashMap<u32, usize> = HashMap::new();
692    let mut total = 0usize;
693
694    for result in parser.records_json_value() {
695        let record = match result {
696            Ok(r) => r,
697            Err(_) => continue,
698        };
699        total += 1;
700        let event_id = record
701            .data
702            .get("Event")
703            .and_then(|e| e.get("System"))
704            .and_then(event_id_from_system)
705            .unwrap_or(0);
706        *counts.entry(event_id).or_insert(0) += 1;
707    }
708
709    let mut by_event_id: Vec<EventFrequency> = counts
710        .into_iter()
711        .map(|(event_id, count)| EventFrequency { event_id, count })
712        .collect();
713    by_event_id.sort_by(|a, b| b.count.cmp(&a.count).then(a.event_id.cmp(&b.event_id)));
714
715    Ok(FrequencyReport {
716        total_events: total,
717        by_event_id,
718    })
719}

723/// Extract every supported semantic event type from an EVTX file and return
724/// them as a timestamp-sorted `Vec<EvtxEvent>`.
725///
726/// Each extractor is called independently; a failure in one (e.g. no matching
727/// EIDs) is silently skipped so the others can still contribute results.
728pub fn extract_all(path: &Path) -> Result<Vec<EvtxEvent>, AnalyzeError> {
729    // Verify the path is accessible before fanning out.
730    let _ = std::fs::metadata(path).map_err(AnalyzeError::Io)?;
731
732    let mut events: Vec<EvtxEvent> = Vec::new();
733
734    macro_rules! push_all {
735        ($fn:ident, $variant:ident) => {
736            if let Ok(items) = $fn(path) {
737                events.extend(items.into_iter().map(EvtxEvent::$variant));
738            }
739        };
740    }
741
742    push_all!(lateral_movement, LateralMovement);
743    push_all!(rdp_sessions, RdpSession);
744    push_all!(smb_access, SmbAccess);
745    push_all!(defender_events, Defender);
746    push_all!(wmi_events, Wmi);
747    push_all!(scheduled_tasks, ScheduledTask);
748    push_all!(process_cmdlines, ProcessExecution);
749
750    events.sort_unstable_by(|a, b| a.timestamp().cmp(b.timestamp()));
751
752    Ok(events)
753}

1222// ── IOC regex patterns ────────────────────────────────────────────────────────
1223
1224struct IocPatterns {
1225    ipv4: regex::Regex,
1226    sha256: regex::Regex,
1227    sha1: regex::Regex,
1228    md5: regex::Regex,
1229    filepath: regex::Regex,
1230}
1231
1232static IOC_PATTERNS: OnceLock<IocPatterns> = OnceLock::new();
1233
1234fn ioc_patterns() -> &'static IocPatterns {
1235    IOC_PATTERNS.get_or_init(|| IocPatterns {
1236        // IPv4 — four dotted octets (0-255 each); reject private-only 127.0.0.1 style
1237        ipv4: regex::Regex::new(
1238            r"\b(?:(?:25[0-5]|2[0-4]\d|[01]?\d\d?)\.){3}(?:25[0-5]|2[0-4]\d|[01]?\d\d?)\b",
1239        )
1240        .unwrap(),
1241        // SHA-256: 64 hex chars
1242        sha256: regex::Regex::new(r"\b[0-9a-fA-F]{64}\b").unwrap(),
1243        // SHA-1: 40 hex chars
1244        sha1: regex::Regex::new(r"\b[0-9a-fA-F]{40}\b").unwrap(),
1245        // MD5: 32 hex chars
1246        md5: regex::Regex::new(r"\b[0-9a-fA-F]{32}\b").unwrap(),
1247        // Windows path: drive letter followed by backslash
1248        filepath: regex::Regex::new(r#"[A-Za-z]:\\[^\s"<>|?*\x00-\x1f]{2,}"#).unwrap(),
1249    })
1250}
1251
1252/// Scan a JSON string for all IOC patterns, returning (kind, value) pairs.
1253///
1254/// The `\b` word-boundary anchors on hex patterns guarantee exact-length
1255/// matching: a 64-char hex string cannot match the 40-char SHA-1 pattern
1256/// because there is no word boundary in the middle of a hex sequence.
1257fn scan_for_iocs(text: &str) -> Vec<(IocKind, String)> {
1258    let p = ioc_patterns();
1259    let mut hits: Vec<(IocKind, String)> = Vec::new();
1260
1261    for m in p.sha256.find_iter(text) {
1262        hits.push((IocKind::Sha256, m.as_str().to_ascii_lowercase()));
1263    }
1264    for m in p.sha1.find_iter(text) {
1265        hits.push((IocKind::Sha1, m.as_str().to_ascii_lowercase()));
1266    }
1267    for m in p.md5.find_iter(text) {
1268        hits.push((IocKind::Md5, m.as_str().to_ascii_lowercase()));
1269    }
1270    for m in p.ipv4.find_iter(text) {
1271        let s = m.as_str().to_owned();
1272        if s != "0.0.0.0" && s != "255.255.255.255" && s != "127.0.0.1" {
1273            hits.push((IocKind::IpAddress, s));
1274        }
1275    }
1276    for m in p.filepath.find_iter(text) {
1277        hits.push((IocKind::FilePath, m.as_str().to_owned()));
1278    }
1279    hits
1280}

1282/// Extract indicators of compromise from all string fields in an EVTX file.
1283///
1284/// Scans IPv4 addresses, MD5/SHA-1/SHA-256 hashes, and Windows
1285/// file paths from every event.  Results are deduplicated and sorted
1286/// by observation count (descending).
1287pub fn ioc_extract(path: &Path) -> Result<IocReport, AnalyzeError> {
1288    let _ = std::fs::metadata(path).map_err(AnalyzeError::Io)?;
1289
1290    let mut parser =
1291        evtx::EvtxParser::from_path(path).map_err(|e| AnalyzeError::Parse(e.to_string()))?;
1292
1293    // (kind, value) → (count, first_ts, last_ts, record_ids)
1294    type Meta = (usize, Option<String>, Option<String>, Vec<u64>);
1295    let mut seen: HashMap<(IocKind, String), Meta> = HashMap::new();
1296    let mut total = 0usize;
1297
1298    for result in parser.records_json_value() {
1299        let record = match result {
1300            Ok(r) => r,
1301            Err(_) => continue,
1302        };
1303        total += 1;
1304        let ts = record.timestamp.to_string();
1305        let record_id = record.event_record_id;
1306
1307        // Serialize EventData to a flat string for pattern scanning
1308        let text = if let Some(ed) = record.data.get("Event").and_then(|e| e.get("EventData")) {
1309            serde_json::to_string(ed).unwrap_or_default()
1310        } else {
1311            serde_json::to_string(&record.data).unwrap_or_default()
1312        };
1313
1314        for (kind, value) in scan_for_iocs(&text) {
1315            let entry = seen
1316                .entry((kind, value))
1317                .or_insert((0, None, None, Vec::new()));
1318            entry.0 += 1;
1319            if entry.1.is_none() {
1320                entry.1 = Some(ts.clone());
1321            }
1322            entry.2 = Some(ts.clone());
1323            if entry.3.len() < 10 {
1324                entry.3.push(record_id);
1325            }
1326        }
1327    }
1328
1329    let mut iocs: Vec<Ioc> = seen
1330        .into_iter()
1331        .map(
1332            |((kind, value), (count, first_seen, last_seen, record_ids))| Ioc {
1333                value,
1334                kind,
1335                count,
1336                first_seen,
1337                last_seen,
1338                record_ids,
1339            },
1340        )
1341        .collect();
1342    iocs.sort_by(|a, b| b.count.cmp(&a.count).then(a.value.cmp(&b.value)));
1343
1344    Ok(IocReport {
1345        events_scanned: total,
1346        iocs,
1347    })
1348}

1581/// Search all string values in every event's JSON blob for a case-insensitive
1582/// substring match.  Returns matching `TimelineEntry` objects.
1583/// Exits with detections semantics (caller should use exit 1 if non-empty).
1584pub fn pivot(path: &Path, query: &str) -> Result<Vec<TimelineEntry>, AnalyzeError> {
1585    let query_lower = query.to_ascii_lowercase();
1586    let _ = std::fs::metadata(path).map_err(AnalyzeError::Io)?;
1587    let mut parser =
1588        evtx::EvtxParser::from_path(path).map_err(|e| AnalyzeError::Parse(e.to_string()))?;
1589
1590    let mut entries = Vec::new();
1591    for result in parser.records_json_value() {
1592        let record = match result {
1593            Ok(r) => r,
1594            Err(_) => continue,
1595        };
1596        if !value_contains_str(&record.data, &query_lower) {
1597            continue;
1598        }
1599        let system = record.data.get("Event").and_then(|e| e.get("System"));
1600        entries.push(TimelineEntry {
1601            record_id: record.event_record_id,
1602            timestamp: record.timestamp.to_string(),
1603            event_id: system.and_then(event_id_from_system).unwrap_or(0),
1604            level: system
1605                .and_then(|s| s.get("Level"))
1606                .and_then(serde_json::Value::as_u64)
1607                .map(|n| n as u8),
1608            channel: system
1609                .and_then(|s| s.get("Channel"))
1610                .and_then(serde_json::Value::as_str)
1611                .map(str::to_owned),
1612            computer: system
1613                .and_then(|s| s.get("Computer"))
1614                .and_then(serde_json::Value::as_str)
1615                .map(str::to_owned),

1617                .and_then(|s| s.get("Provider"))
1618                .and_then(|p| p.get("@Name").or_else(|| p.get("@Guid")))

1626fn value_contains_str(v: &serde_json::Value, query: &str) -> bool {
1627    match v {
1628        serde_json::Value::String(s) => s.to_ascii_lowercase().contains(query),
1629        serde_json::Value::Array(arr) => arr.iter().any(|item| value_contains_str(item, query)),
1630        serde_json::Value::Object(map) => {
1631            map.values().any(|val| value_contains_str(val, query))
1632        }
1633        _ => false,
1634    }
1635}

1637fn value_matches_regex(v: &serde_json::Value, re: &regex::Regex) -> bool {
1638    match v {
1639        serde_json::Value::String(s) => re.is_match(s),
1640        serde_json::Value::Array(arr) => arr.iter().any(|item| value_matches_regex(item, re)),
1641        serde_json::Value::Object(map) => map.values().any(|val| value_matches_regex(val, re)),
1642        _ => false,
1643    }
1644}

1646/// Search all event string values for a substring or regex pattern.
1647///
1648/// When `use_regex` is false, performs the same case-insensitive substring
1649/// match as the former `pivot` command.  When true, compiles `query` as a
1650/// regex and applies it to every string value in the event JSON.
1651///
1652/// Returns `Err(AnalyzeError::Parse)` when `use_regex` is true and `query`
1653/// is not a valid regex.

1665    let mut parser =
1666        evtx::EvtxParser::from_path(path).map_err(|e| AnalyzeError::Parse(e.to_string()))?;
1667    let mut entries = Vec::new();
1668    for result in parser.records_json_value() {
1669        let record = match result {
1670            Ok(r) => r,
1671            Err(_) => continue,
1672        };
1673        if !value_matches_regex(&record.data, &re) {
1674            continue;
1675        }
1676        let system = record.data.get("Event").and_then(|e| e.get("System"));
1677        entries.push(TimelineEntry {
1678            record_id: record.event_record_id,
1679            timestamp: record.timestamp.to_string(),
1680            event_id: system.and_then(event_id_from_system).unwrap_or(0),
1681            level: system
1682                .and_then(|s| s.get("Level"))
1683                .and_then(serde_json::Value::as_u64)
1684                .map(|n| n as u8),
1685            channel: system
1686                .and_then(|s| s.get("Channel"))
1687                .and_then(serde_json::Value::as_str)
1688                .map(str::to_owned),
1689            computer: system
1690                .and_then(|s| s.get("Computer"))
1691                .and_then(serde_json::Value::as_str)
1692                .map(str::to_owned),
1693            provider: system
1694                .and_then(|s| s.get("Provider"))
1695                .and_then(|p| p.get("@Name").or_else(|| p.get("@Guid")))
1696                .and_then(serde_json::Value::as_str)
1697                .map(str::to_owned),
1698        });
1699    }
1700    Ok(entries)
1701}

1703/// Compare two EVTX files by record ID.  Returns added (in B not A) and
1704/// removed (in A not B) entries.  Exit 1 when the diff is non-empty.
1705pub fn diff(a: &Path, b: &Path) -> Result<EvtxDiff, AnalyzeError> {
1706    let entries_a = timeline(a)?;
1707    let entries_b = timeline(b)?;
1708    let ids_a: std::collections::HashSet<u64> =
1709        entries_a.iter().map(|e| e.record_id).collect();
1710    let ids_b: std::collections::HashSet<u64> =
1711        entries_b.iter().map(|e| e.record_id).collect();
1712    let added = entries_b
1713        .into_iter()
1714        .filter(|e| !ids_a.contains(&e.record_id))
1715        .collect();
1716    let removed = entries_a
1717        .into_iter()
1718        .filter(|e| !ids_b.contains(&e.record_id))
1719        .collect();
1720    Ok(EvtxDiff { added, removed })
1721}

1723/// Extract process-creation events (Security EID 4688 and Sysmon EID 1) and
1724/// return a flat list of `ProcessNode` records with PID/PPID/image/cmdline.
1725pub fn process_tree(path: &Path) -> Result<Vec<ProcessNode>, AnalyzeError> {
1726    let _ = std::fs::metadata(path).map_err(AnalyzeError::Io)?;
1727    let mut parser =
1728        evtx::EvtxParser::from_path(path).map_err(|e| AnalyzeError::Parse(e.to_string()))?;
1729
1730    let mut nodes = Vec::new();
1731    for result in parser.records_json_value() {
1732        let record = match result {
1733            Ok(r) => r,
1734            Err(_) => continue,
1735        };
1736        let system = record.data.get("Event").and_then(|e| e.get("System"));
1737        let event_id = system.and_then(event_id_from_system).unwrap_or(0);
1738        if event_id != 4688 && event_id != 1 {
1739            continue;
1740        }
1741        let ed = match record.data.get("Event").and_then(|e| e.get("EventData")) {
1742            Some(d) => d,
1743            None => continue,
1744        };
1745        let (pid, parent_pid, image, command_line) = if event_id == 4688 {
1746            let pid = event_data_str(ed, "NewProcessId")
1747                .and_then(|s| u64::from_str_radix(s.trim_start_matches("0x"), 16).ok())
1748                .unwrap_or(0);
1749            let ppid = event_data_str(ed, "ProcessId")
1750                .and_then(|s| u64::from_str_radix(s.trim_start_matches("0x"), 16).ok())
1751                .unwrap_or(0);
1752            let image = event_data_str(ed, "NewProcessName").unwrap_or("-").to_owned();
1753            let cmdline = event_data_str(ed, "CommandLine").unwrap_or("").to_owned();
1754            (pid, ppid, image, cmdline)
1755        } else {
1756            // Sysmon EID 1 — flat JSON object format, integer PIDs
1757            let pid = sysmon_pid(ed, "ProcessId");
1758            let ppid = sysmon_pid(ed, "ParentProcessId");
1759            let image = sysmon_str(ed, "Image").unwrap_or("-").to_owned();
1760            let cmdline = sysmon_str(ed, "CommandLine").unwrap_or("").to_owned();
1761            (pid, ppid, image, cmdline)
1762        };
1763        nodes.push(ProcessNode {
1764            pid,
1765            parent_pid,
1766            image,
1767            command_line,
1768            timestamp: record.timestamp.to_string(),
1769        });
1770    }
1771    Ok(nodes)

1774/// Resolve the true *source* machine from an EID 4624 event's fields.
1775///
1776/// For Logon Type 10 (RDP with NLA disabled), `WorkstationName` is written by
1777/// Windows as the *destination* (the machine being accessed), not the source.
1778/// `IpAddress` (SourceNetworkAddress) is the reliable source for all logon types.
1779/// For all other logon types, `WorkstationName` is the source and is preferred
1780/// over the IP because it gives the hostname.
1781///
1782/// Citation: Ahmed Thabit & Ahmed Abdo, "Be careful when interpreting Windows
1783/// event fields" (2025).
1784/// <https://www.linkedin.com/posts/mr-ahmed-thabit_be-careful-when-interpreting-windows-event-activity-7461407456984772608-Okyl>
1785///
1786/// Returns `None` when no usable source can be identified (skip the event).
1787fn resolve_logon_source(logon_type: u32, workstation: &str, ip: &str) -> Option<String> {
1788    let ip_usable = !ip.is_empty() && ip != "-" && ip != "::1" && ip != "127.0.0.1";
1789    if logon_type == 10 {
1790        // RDP (RemoteInteractive): WorkstationName = destination; only trust IpAddress.
1791        ip_usable.then(|| ip.to_owned())
1792    } else {
1793        let ws_usable = !workstation.is_empty() && workstation != "-";
1794        if ws_usable {
1795            Some(workstation.to_owned())
1796        } else if ip_usable {
1797            Some(ip.to_owned())
1798        } else {
1799            None
1800        }
1801    }
1802}

1804/// Build a logon source→target graph from EID 4624 events.
1805pub fn logon_graph(path: &Path) -> Result<LogonGraph, AnalyzeError> {
1806    let _ = std::fs::metadata(path).map_err(AnalyzeError::Io)?;
1807    let mut parser =
1808        evtx::EvtxParser::from_path(path).map_err(|e| AnalyzeError::Parse(e.to_string()))?;
1809
1810    let mut edge_map: HashMap<(String, String, u32), usize> = HashMap::new();
1811
1812    for result in parser.records_json_value() {
1813        let record = match result {
1814            Ok(r) => r,
1815            Err(_) => continue,
1816        };
1817        let system = record.data.get("Event").and_then(|e| e.get("System"));
1818        if system.and_then(event_id_from_system) != Some(4624) {
1819            continue;
1820        }
1821        let ed = match record.data.get("Event").and_then(|e| e.get("EventData")) {
1822            Some(d) => d,
1823            None => continue,
1824        };
1825        let computer = system
1826            .and_then(|s| s.get("Computer"))
1827            .and_then(serde_json::Value::as_str)
1828            .unwrap_or("-")
1829            .to_owned();
1830        let workstation =
1831            event_data_str(ed, "WorkstationName").unwrap_or("").to_owned();
1832        let ip = event_data_str(ed, "IpAddress").unwrap_or("").to_owned();
1833        let logon_type: u32 = event_data_str(ed, "LogonType")
1834            .and_then(|s| s.parse().ok())
1835            .unwrap_or(0);
1836
1837        let source = match resolve_logon_source(logon_type, &workstation, &ip) {
1838            Some(s) => s,
1839            None => continue,
1840        };
1841
1842        *edge_map.entry((source, computer, logon_type)).or_insert(0) += 1;
1843    }
1844
1845    let mut node_set = std::collections::HashSet::new();
1846    let mut edges = Vec::new();
1847    for ((source, target, logon_type), count) in edge_map {
1848        node_set.insert(source.clone());
1849        node_set.insert(target.clone());
1850        edges.push(LogonEdge { source, target, logon_type, count });
1851    }
1852
1853    let mut nodes: Vec<String> = node_set.into_iter().collect();
1854    nodes.sort();
1855    edges.sort_by(|a, b| a.source.cmp(&b.source));
1856
1857    Ok(LogonGraph { nodes, edges })
1858}

2019/// Detect and decode a PowerShell `-EncodedCommand` (or `-enc` / `-ec`) payload.
2020///
2021/// Windows PowerShell encodes commands as Base64 of UTF-16LE bytes.
2022/// Returns `Some(decoded)` when detected, `None` for plain scripts.
2023pub fn deobfuscate_ps(text: &str) -> Option<String> {
2024    use base64::{Engine as _, engine::general_purpose::STANDARD};
2025
2026    let flags = ["-encodedcommand", "-enc", "-ec", "-en"];
2027    let tokens: Vec<&str> = text.split_whitespace().collect();
2028    for (i, tok) in tokens.iter().enumerate() {
2029        if flags.contains(&tok.to_ascii_lowercase().as_str()) {
2030            if let Some(payload) = tokens.get(i + 1) {
2031                if let Ok(bytes) = STANDARD.decode(payload) {
2032                    // Interpret as UTF-16LE
2033                    let utf16: Vec<u16> = bytes
2034                        .chunks_exact(2)
2035                        .map(|c| u16::from_le_bytes([c[0], c[1]]))
2036                        .collect();
2037                    return Some(String::from_utf16_lossy(&utf16));
2038                }
2039            }
2040        }
2041    }
2042    None
2043}

2045// ── Frequency anomaly scoring ─────────────────────────────────────────────────
2046
2047/// One event ID's anomaly record.
2048#[derive(Debug, serde::Serialize)]
2049pub struct AnomalyEntry {
2050    pub event_id: u32,
2051    pub count: usize,
2052    pub z_score: f64,
2053}
2054
2055/// Compute a z-score frequency anomaly for every event ID in `path`.
2056///
2057/// Returns all entries with `|z_score| >= min_z`, sorted descending by `|z_score|`.
2058/// A `min_z` of `0.0` returns all event IDs.
2059pub fn anomaly(path: &Path, min_z: f64) -> Result<Vec<AnomalyEntry>, AnalyzeError> {
2060    let report = frequency(path)?;
2061    let counts: Vec<f64> = report.by_event_id.iter().map(|f| f.count as f64).collect();
2062    if counts.is_empty() {
2063        return Ok(vec![]);
2064    }
2065    let mean = counts.iter().sum::<f64>() / counts.len() as f64;
2066    let variance = counts.iter().map(|c| (c - mean).powi(2)).sum::<f64>() / counts.len() as f64;
2067    let std_dev = variance.sqrt();
2068
2069    let mut entries: Vec<AnomalyEntry> = report
2070        .by_event_id
2071        .iter()
2072        .map(|f| {
2073            let z = if std_dev > 0.0 {
2074                (f.count as f64 - mean) / std_dev
2075            } else {
2076                0.0
2077            };
2078            AnomalyEntry { event_id: f.event_id, count: f.count, z_score: z }
2079        })
2080        .filter(|e| e.z_score.abs() >= min_z)
2081        .collect();
2082
2083    entries.sort_by(|a, b| b.z_score.abs().partial_cmp(&a.z_score.abs()).unwrap_or(std::cmp::Ordering::Equal));
2084    Ok(entries)
2085}

2350/// Extract WMI provider/subscription events from EID 5857, 5858, 5860, 5861.
2351pub fn wmi_events(path: &Path) -> Result<Vec<WmiEvent>, AnalyzeError> {
2352    let _ = std::fs::metadata(path).map_err(AnalyzeError::Io)?;
2353    let mut parser =
2354        evtx::EvtxParser::from_path(path).map_err(|e| AnalyzeError::Parse(e.to_string()))?;
2355
2356    let mut events = Vec::new();
2357    for result in parser.records_json_value() {
2358        let record = match result {
2359            Ok(r) => r,
2360            Err(_) => continue,
2361        };
2362        let system = record.data.get("Event").and_then(|e| e.get("System"));
2363        let event_id = match system.and_then(event_id_from_system) {
2364            // 5857/5858/5860/5861 = WMI-Activity operational log
2365            // 19/20/21 = Sysmon WMI persistence (WmiEventFilter/Consumer/ConsumerToFilter)
2366            Some(id) if matches!(id, 5857 | 5858 | 5860 | 5861 | 19 | 20 | 21) => id,
2367            _ => continue,
2368        };
2369        let event_node = record.data.get("Event");
2370        // WMI-Activity events (EID 5860/5861) use UserData → Operation_*
2371        // instead of EventData.  Fall back to the first child of UserData
2372        // when EventData is absent so consumer/filter fields are populated.
2373        let ed = event_node.and_then(|e| e.get("EventData")).or_else(|| {
2374            event_node
2375                .and_then(|e| e.get("UserData"))
2376                .and_then(|ud| ud.as_object())
2377                .and_then(|obj| obj.values().find(|v| v.is_object()))
2378        });
2379        let provider = system
2380            .and_then(|s| s.get("Provider"))
2381            .and_then(|p| p.get("@Name").or_else(|| p.get("@Guid")))
2382            .and_then(serde_json::Value::as_str)
2383            .map(str::to_owned);
2384
2385        let (filter_name, consumer_name, query) = if let Some(ed) = ed {
2386            (
2387                // EID 5861 UserData: "ESS"; Sysmon EID 19: "Name"; EID 21: "Filter"
2388                event_data_str(ed, "ESS")
2389                    .or_else(|| event_data_str(ed, "FilterName"))
2390                    .or_else(|| event_data_str(ed, "PossibleCause"))
2391                    .or_else(|| {
2392                        // Sysmon EID 19 (WmiEventFilter) and EID 21 use "Name"/"Filter"
2393                        if matches!(event_id, 19 | 21) {
2394                            event_data_str(ed, "Name")
2395                                .or_else(|| event_data_str(ed, "Filter"))
2396                        } else {
2397                            None
2398                        }
2399                    })
2400                    .map(str::to_owned),
2401                // EID 5861 UserData: "CONSUMER"; Sysmon EID 20: "Name"; EID 21: "Consumer"
2402                event_data_str(ed, "CONSUMER")
2403                    .or_else(|| event_data_str(ed, "ConsumerName"))
2404                    .or_else(|| {
2405                        if matches!(event_id, 20 | 21) {
2406                            event_data_str(ed, "Name")
2407                                .or_else(|| event_data_str(ed, "Consumer"))
2408                        } else {
2409                            None
2410                        }
2411                    })
2412                    .map(str::to_owned),
2413                event_data_str(ed, "Query").map(str::to_owned),
2414            )
2415        } else {
2416            (None, None, None)
2417        };
2418
2419        events.push(WmiEvent {
2420            timestamp: record.timestamp.to_string(),
2421            event_id,
2422            provider,
2423            filter_name,
2424            consumer_name,
2425            query,
2426        });
2427    }
2428    Ok(events)
2429}

2486/// Extract process command lines from Security EID 4688 and Sysmon EID 1,
2487/// with LOLBin tagging.
2488pub fn process_cmdlines(path: &Path) -> Result<Vec<ProcessExecution>, AnalyzeError> {
2489    let _ = std::fs::metadata(path).map_err(AnalyzeError::Io)?;
2490    let mut parser =
2491        evtx::EvtxParser::from_path(path).map_err(|e| AnalyzeError::Parse(e.to_string()))?;
2492
2493    let mut execs = Vec::new();
2494    for result in parser.records_json_value() {
2495        let record = match result {
2496            Ok(r) => r,
2497            Err(_) => continue,
2498        };
2499        let system = record.data.get("Event").and_then(|e| e.get("System"));
2500        let event_id = match system.and_then(event_id_from_system) {
2501            Some(id @ (4688 | 1)) => id,
2502            _ => continue,
2503        };
2504        let ed = match record.data.get("Event").and_then(|e| e.get("EventData")) {
2505            Some(d) => d,
2506            None => continue,
2507        };
2508        let (image, command_line, pid, parent_pid, parent_image) = if event_id == 4688 {
2509            let image = event_data_str(ed, "NewProcessName").unwrap_or("-").to_owned();
2510            let cmdline = event_data_str(ed, "CommandLine").unwrap_or("").to_owned();
2511            let pid = event_data_str(ed, "NewProcessId")
2512                .and_then(|s| u64::from_str_radix(s.trim_start_matches("0x"), 16).ok())
2513                .unwrap_or(0);
2514            let ppid = event_data_str(ed, "ProcessId")
2515                .and_then(|s| u64::from_str_radix(s.trim_start_matches("0x"), 16).ok())
2516                .unwrap_or(0);
2517            let parent_image = event_data_str(ed, "ParentProcessName").map(str::to_owned);
2518            (image, cmdline, pid, ppid, parent_image)
2519        } else {
2520            // Sysmon EID 1 — flat JSON object format, integer PIDs
2521            let image = sysmon_str(ed, "Image").unwrap_or("-").to_owned();
2522            let cmdline = sysmon_str(ed, "CommandLine").unwrap_or("").to_owned();
2523            let pid = sysmon_pid(ed, "ProcessId");
2524            let ppid = sysmon_pid(ed, "ParentProcessId");
2525            let parent_image = sysmon_str(ed, "ParentImage").map(str::to_owned);
2526            (image, cmdline, pid, ppid, parent_image)
2527        };
2528        let lolbin = is_lolbin(&image);
2529
2530        execs.push(ProcessExecution {
2531            timestamp: record.timestamp.to_string(),
2532            event_id,
2533            pid,
2534            parent_pid,
2535            image,
2536            command_line,
2537            parent_image,
2538            is_lolbin: lolbin,
2539        });
2540    }
2541    Ok(execs)
2542}
