1//! Template instance decoding.
2//!
3//! A real EVTX record's payload is almost always a single template instance: a
4//! reference to a reusable definition (the System/EventData *structure* with
5//! substitution placeholders) plus the per-record substitution *values*. This
6//! module reads the instance header + values, then parses the definition body
7//! with those values in scope so the deserializer resolves placeholders inline.
8//!
9//! Correctness first: the definition body is re-parsed per instance (a GUID
10//! cache is a later optimization). Array and embedded-BinXml substitution values
11//! are not yet handled (scalar values only).
12

21/// Allocation cap on the substitution-value count of one instance.
22const MAX_SUBSTITUTIONS: usize = 4096;
23
24/// Read a template instance (the bytes after the `0x0c` token) and return the
25/// resolved content nodes. `chunk` is the full chunk slice (the addressing base
26/// for the definition + its names).
27pub(crate) fn read_template_instance(
28    cur: &mut Cursor<'_>,
29    chunk: &[u8],
30    names: &mut NameCache,
31    limits: Limits,
32) -> Result<Vec<Node>, DeserializeError> {
33    // Instance header: unused u8, template_id u32, definition offset u32.
34    let _unused = cur.read_u8()?;
35    let _template_id = cur.read_u32_le()?;
36    let def_offset = cur.read_u32_le()? as usize;
37
38    // The definition is either present inline here (first use in the chunk) or
39    // already written elsewhere in the chunk (a back-reference).
40    let (body_offset, body_len) = if cur.position() == def_offset {
41        let data_size = read_def_header(cur)?;
42        let body_offset = cur.position();
43        cur.skip(data_size)?; // skip past the inline definition body
44        (body_offset, data_size)
45    } else {
46        let mut at_def = Cursor::at(chunk, def_offset);
47        let data_size = read_def_header(&mut at_def)?;
48        (at_def.position(), data_size)
49    };
50
51    // Substitution value array: count, then count×(size u16, type u8, unused u8),
52    // then the raw values back-to-back.
53    let count = cur.read_u32_le()? as usize;
54    if count > MAX_SUBSTITUTIONS {
55        return Err(DeserializeError::Unsupported("too many substitutions"));
56    }
57    let mut descriptors = Vec::with_capacity(count);
58    for _ in 0..count {
59        let size = cur.read_u16_le()? as usize;
60        let value_type = cur.read_u8()?;
61        let _unused = cur.read_u8()?;
62        descriptors.push((size, value_type));
63    }
64    let mut values = Vec::with_capacity(count);
65    for (size, value_type) in &descriptors {
66        if *value_type == VT_BINXML {
67            // Embedded BinXml: the `size` value bytes are themselves a fragment
68            // (commonly the entire EventData). Decode it against the chunk, then
69            // step the main cursor past the value.
70            let mut frag = Cursor::at(chunk, cur.position());
71            let nodes = deserialize_fragment(&mut frag, chunk, names)?;
72            cur.skip(*size)?;
73            values.push(SubstitutionValue::Nodes(nodes));
74        } else {
75            // Scalar value (array types still surface as ValueError::Unsupported).
76            values.push(SubstitutionValue::Scalar(read_value(cur, *value_type, Some(*size))?));
77        }
78    }
79
80    // Parse the definition body with the values in scope so placeholders resolve
81    // inline. Names resolve against the full chunk; the cursor is bounded to the
82    // body's declared end.
83    let body_end = body_offset
84        .checked_add(body_len)
85        .filter(|end| *end <= chunk.len())
86        .ok_or(DeserializeError::Unsupported("template body out of bounds"))?;
87    let mut body_cur = Cursor::at(chunk, body_offset);
88    run(&mut body_cur, chunk, names, Some(&values), body_end, limits)
89}

91/// Read a definition header (`next u32, guid[16], data_size u32`) and return the
92/// definition body's `data_size`.
93fn read_def_header(cur: &mut Cursor<'_>) -> Result<usize, DeserializeError> {
94    let _next = cur.read_u32_le()?;
95    let _guid = cur.take(16)?;
96    Ok(cur.read_u32_le()? as usize)
97}

105    fn name_struct(s: &str) -> Vec<u8> {
106        let units: Vec<u16> = s.encode_utf16().collect();
107        let mut v = Vec::new();
108        v.extend_from_slice(&0u32.to_le_bytes()); // next_string
109        v.extend_from_slice(&0u16.to_le_bytes()); // hash
110        v.extend_from_slice(&(units.len() as u16).to_le_bytes()); // char_count
111        for u in &units {
112            v.extend_from_slice(&u.to_le_bytes());
113        }
114        v.extend_from_slice(&0u16.to_le_bytes()); // NUL
115        v
116    }

120        buf.extend_from_slice(&struct_start.to_le_bytes());
121        buf.extend_from_slice(&name_struct(name));
122    }

124    /// Build a record whose payload is one template instance:
125    /// definition body `<Event>{subst 0}</Event>`, with `subst[0]` a string.
126    fn build_record(value: &str) -> Vec<u8> {
127        let mut b = Vec::new();
128        // top-level fragment header
129        b.extend_from_slice(&[0x0f, 0x01, 0x01, 0x00]);
130        // template instance
131        b.push(0x0c);
132        b.push(0x00); // unused
133        b.extend_from_slice(&1u32.to_le_bytes()); // template_id
134        let def_off_pos = b.len();
135        b.extend_from_slice(&0u32.to_le_bytes()); // def_offset (patched below)
136        let def_offset = b.len() as u32; // inline: definition header starts here
137        b[def_off_pos..def_off_pos + 4].copy_from_slice(&def_offset.to_le_bytes());
138        // definition header: next u32, guid[16], data_size u32
139        b.extend_from_slice(&0u32.to_le_bytes()); // next
140        b.extend_from_slice(&[0u8; 16]); // guid
141        let data_size_pos = b.len();
142        b.extend_from_slice(&0u32.to_le_bytes()); // data_size (patched)
143        let body_start = b.len();
144        // definition body (template-definition mode → open element carries dep_id):
145        b.extend_from_slice(&[0x0f, 0x01, 0x01, 0x00]); // fragment header
146        b.push(0x01); // open start element, no attrs
147        b.extend_from_slice(&0u16.to_le_bytes()); // dependency_id
148        b.extend_from_slice(&0u32.to_le_bytes()); // element data_size
149        push_inline_name(&mut b, "Event");
150        b.push(0x02); // close start element
151        b.push(0x0d); // normal substitution
152        b.extend_from_slice(&0u16.to_le_bytes()); // index 0
153        b.push(0x01); // value_type String
154        b.push(0x04); // end element
155        b.push(0x00); // end of stream (definition body)
156        let body_len = (b.len() - body_start) as u32;
157        b[data_size_pos..data_size_pos + 4].copy_from_slice(&body_len.to_le_bytes());
158        // substitution value array
159        b.extend_from_slice(&1u32.to_le_bytes()); // count
160        let units: Vec<u16> = value.encode_utf16().collect();
161        let byte_len = (units.len() * 2) as u16;
162        b.extend_from_slice(&byte_len.to_le_bytes()); // size
163        b.push(0x01); // value_type String
164        b.push(0x00); // unused
165        for u in &units {
166            b.extend_from_slice(&u.to_le_bytes()); // sized value, no length prefix
167        }
168        // top-level end of stream
169        b.push(0x00);
170        b
171    }

209    /// Generalized builder: definition body `<Event>{subst}</Event>` where the
210    /// def-body substitution is `subst_token` (0x0d/0x0e) with declared
211    /// `decl_type`, and the single instance value has `value_type` + `value_bytes`.
212    fn build_subst(subst_token: u8, decl_type: u8, value_type: u8, value_bytes: &[u8]) -> Vec<u8> {
213        let mut b = Vec::new();
214        b.extend_from_slice(&[0x0f, 0x01, 0x01, 0x00]);
215        b.push(0x0c);
216        b.push(0x00);
217        b.extend_from_slice(&1u32.to_le_bytes());
218        let def_off_pos = b.len();
219        b.extend_from_slice(&0u32.to_le_bytes());
220        let def_offset = b.len() as u32;
221        b[def_off_pos..def_off_pos + 4].copy_from_slice(&def_offset.to_le_bytes());
222        b.extend_from_slice(&0u32.to_le_bytes());
223        b.extend_from_slice(&[0u8; 16]);
224        let data_size_pos = b.len();
225        b.extend_from_slice(&0u32.to_le_bytes());
226        let body_start = b.len();
227        b.extend_from_slice(&[0x0f, 0x01, 0x01, 0x00]);
228        b.push(0x01);
229        b.extend_from_slice(&0u16.to_le_bytes());
230        b.extend_from_slice(&0u32.to_le_bytes());
231        push_inline_name(&mut b, "Event");
232        b.push(0x02);
233        b.push(subst_token);
234        b.extend_from_slice(&0u16.to_le_bytes()); // index 0
235        b.push(decl_type);
236        b.push(0x04);
237        b.push(0x00);
238        let body_len = (b.len() - body_start) as u32;
239        b[data_size_pos..data_size_pos + 4].copy_from_slice(&body_len.to_le_bytes());
240        // one instance value
241        b.extend_from_slice(&1u32.to_le_bytes()); // count
242        b.extend_from_slice(&(value_bytes.len() as u16).to_le_bytes());
243        b.push(value_type);
244        b.push(0x00); // unused
245        b.extend_from_slice(value_bytes);
246        b.push(0x00);
247        b
248    }

370    /// Append a direct-mode BinXml fragment `<name>value</name>` (used as an
371    /// embedded-BinXml substitution value; inline name offsets are absolute).
372    fn push_embedded_fragment(b: &mut Vec<u8>, name: &str, value: &str) {
373        b.extend_from_slice(&[0x0f, 0x01, 0x01, 0x00]);
374        b.push(0x01); // open, no attrs, no dep_id (direct mode)
375        b.extend_from_slice(&0u32.to_le_bytes());
376        push_inline_name(b, name);
377        b.push(0x02);
378        b.push(0x05);
379        b.push(0x01);
380        let units: Vec<u16> = value.encode_utf16().collect();
381        b.extend_from_slice(&(units.len() as u16).to_le_bytes());
382        for u in &units {
383            b.extend_from_slice(&u.to_le_bytes());
384        }
385        b.push(0x04);
386        b.push(0x00);
387    }
