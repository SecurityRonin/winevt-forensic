1//! The in-memory tree (IR) produced by the BinXml deserializer.
2//!
3//! A decoded fragment is a list of [`Node`]s; an [`Element`] owns its
4//! attributes and child nodes. The tree is intentionally simple and owned
5//! (no borrows into the chunk) so it survives past the decode and can be
6//! projected to JSON/XML independently.
7
8/// A node in the decoded tree: either an element or a run of text.
9#[derive(Debug, Clone, PartialEq)]
10pub enum Node {
11    /// An XML element with a name, attributes and children.
12    Element(Element),
13    /// Character data / a substituted value rendered to text.
14    Text(String),
15}
16
17/// An XML element.
18#[derive(Debug, Clone, PartialEq, Default)]
19pub struct Element {
20    /// The element (tag) name.
21    pub name: String,
22    /// Attributes as `(name, value)` pairs, in document order.
23    pub attributes: Vec<(String, String)>,
24    /// Child nodes, in document order.
25    pub children: Vec<Node>,
26}
