1//! Container glue: iterate an EVTX file's chunks and records and decode each
2//! record's BinXml payload into a flat [`DecodedRecord`].
3//!
4//! Names and templates are chunk-scoped, so a fresh [`NameCache`] is used per
5//! chunk and the whole 64 KiB chunk is the addressing base for every record in
6//! it. A record whose payload fails to decode (e.g. it uses a substitution type
7//! not yet supported) is skipped, not fatal — the rest of the file still
8//! decodes.
9
10#![allow(clippy::doc_markdown)] // "BinXml"/"EventData" appear throughout these docs
11
12use winevt_core::binary::{
13    EvtxChunkHeader, EvtxFileHeader, EvtxRecordHeader, CHUNK_RECORDS_OFFSET, CHUNK_SIZE,
14};
15
16use crate::cursor::Cursor;
17use crate::deserializer::deserialize_fragment;
18use crate::extract::{extract_record, DecodedRecord};
19use crate::name::NameCache;

21/// The EVTX file header block is 4 KiB; the first chunk starts after it.
22const FILE_HEADER_BLOCK: usize = 0x1000;
23/// Bytes of fixed record header before the BinXml payload.
24const RECORD_HEADER_SIZE: usize = 24;
25
26/// One decoded record with its container-level identity.
27#[derive(Debug, Clone, PartialEq, Eq)]
28pub struct RecordEntry {
29    /// Event record id from the record header.
30    pub record_id: u64,
31    /// Record-header timestamp (Windows FILETIME, 100 ns since 1601).
32    pub timestamp_filetime: u64,
33    /// The decoded record content.
34    pub record: DecodedRecord,
35}
36
37/// Decode every record in an in-memory EVTX file. Returns an empty vec if the
38/// file header is not a valid `ElfFile`. Undecodable records are skipped.
39#[must_use]
40pub fn decode_file(data: &[u8]) -> Vec<RecordEntry> {
41    let mut out = Vec::new();
42    if EvtxFileHeader::parse(data).is_none() {
43        return out;
44    }
45    let chunk_size = usize::try_from(CHUNK_SIZE).unwrap_or(usize::MAX);
46    let mut chunk_off = FILE_HEADER_BLOCK;
47    while let Some(chunk) = data.get(chunk_off..chunk_off.saturating_add(chunk_size)) {
48        if let Some(header) = EvtxChunkHeader::parse(chunk) {
49            decode_chunk(chunk, header.free_space_offset as usize, &mut out);
50        }
51        chunk_off = chunk_off.saturating_add(chunk_size);
52    }
53    out
54}
55
56/// Decode the records of one chunk (a fresh name cache scoped to this chunk).
57fn decode_chunk(chunk: &[u8], free_space_offset: usize, out: &mut Vec<RecordEntry>) {
58    let mut names = NameCache::new();
59    let end = free_space_offset.min(chunk.len());
60    let mut rec_off = usize::try_from(CHUNK_RECORDS_OFFSET).unwrap_or(usize::MAX);
61    while rec_off.saturating_add(RECORD_HEADER_SIZE) <= end {
62        let Some(header) = chunk.get(rec_off..).and_then(EvtxRecordHeader::parse) else {
63            break;
64        };
65        let size = header.size as usize;
66        let payload_end = match rec_off.checked_add(size) {
67            Some(e) if size >= RECORD_HEADER_SIZE && e <= chunk.len() => e,
68            _ => break,
69        };
70        let binxml_off = rec_off + RECORD_HEADER_SIZE;
71        let mut cur = Cursor::at(chunk, binxml_off);
72        if let Ok(nodes) = deserialize_fragment(&mut cur, chunk, &mut names) {
73            out.push(RecordEntry {
74                record_id: header.record_id,
75                timestamp_filetime: header.timestamp,
76                record: extract_record(&nodes),
77            });
78        }
79        rec_off = payload_end;
80    }
81}

87    #[test]
88    fn non_evtx_input_yields_no_records() {
89        assert!(decode_file(b"not an evtx file at all").is_empty());
90        assert!(decode_file(&[]).is_empty());

93    /// A minimal single-record EVTX file: one chunk, one record whose payload is
94    /// the template-free fragment `<Event/>`. (The real-fixture differential
95    /// parity test lives in `tests/real_corpus_parity.rs`.)
96    fn synthetic_evtx() -> Vec<u8> {
97        let cs = usize::try_from(CHUNK_SIZE).unwrap();
98        let mut chunk = vec![0u8; cs];
99        chunk[..8].copy_from_slice(b"ElfChnk\0");
100        let rec_off = usize::try_from(CHUNK_RECORDS_OFFSET).unwrap();
101        let mut p = rec_off;
102        chunk[p..p + 4].copy_from_slice(&[0x2a, 0x2a, 0x00, 0x00]); // record magic
103        p += 4;
104        let size_at = p;
105        p += 4; // size (patched)
106        chunk[p..p + 8].copy_from_slice(&7u64.to_le_bytes()); // record_id
107        p += 8;
108        chunk[p..p + 8].copy_from_slice(&0u64.to_le_bytes()); // timestamp
109        p += 8;
110        // BinXml `<Event/>` (direct fragment); p is now rec_off + 24.
111        chunk[p..p + 4].copy_from_slice(&[0x0f, 0x01, 0x01, 0x00]); // fragment header
112        p += 4;
113        chunk[p] = 0x01; // open start element, no attrs / dep id
114        p += 1;
115        chunk[p..p + 4].copy_from_slice(&0u32.to_le_bytes()); // element data_size
116        p += 4;
117        let name_off = u32::try_from(p + 4).unwrap(); // inline: struct follows the u32
118        chunk[p..p + 4].copy_from_slice(&name_off.to_le_bytes());
119        p += 4;
120        chunk[p..p + 4].copy_from_slice(&0u32.to_le_bytes()); // next_string
121        p += 4;
122        chunk[p..p + 2].copy_from_slice(&0u16.to_le_bytes()); // hash
123        p += 2;
124        let chars: Vec<u16> = "Event".encode_utf16().collect();
125        chunk[p..p + 2].copy_from_slice(&u16::try_from(chars.len()).unwrap().to_le_bytes());
126        p += 2;
127        for c in &chars {
128            chunk[p..p + 2].copy_from_slice(&c.to_le_bytes());
129            p += 2;
130        }
131        chunk[p..p + 2].copy_from_slice(&0u16.to_le_bytes()); // NUL terminator
132        p += 2;
133        chunk[p] = 0x03; // close empty element
134        p += 1;
135        chunk[p] = 0x00; // end of stream
136        p += 1;
137        chunk[p..p + 4].copy_from_slice(&0u32.to_le_bytes()); // size copy
138        p += 4;
139        let rec_size = u32::try_from(p - rec_off).unwrap();
140        chunk[size_at..size_at + 4].copy_from_slice(&rec_size.to_le_bytes());
141        chunk[48..52].copy_from_slice(&u32::try_from(p).unwrap().to_le_bytes()); // free_space_offset
142        chunk[44..48].copy_from_slice(&u32::try_from(rec_off).unwrap().to_le_bytes());
143        let mut file = vec![0u8; FILE_HEADER_BLOCK];
144        file[..8].copy_from_slice(b"ElfFile\0");
145        file.extend_from_slice(&chunk);
146        file
147    }

149    #[test]
150    fn decodes_a_synthetic_record() {
151        let records = decode_file(&synthetic_evtx());
152        assert_eq!(records.len(), 1);
153        assert_eq!(records[0].record_id, 7);
154        assert_eq!(records[0].record.event_id, 0); // <Event/> has no System
155    }

157    #[test]
158    fn bad_record_magic_stops_the_chunk() {
159        let mut data = synthetic_evtx();
160        let rec = FILE_HEADER_BLOCK + usize::try_from(CHUNK_RECORDS_OFFSET).unwrap();
161        data[rec] = 0xFF; // corrupt the record magic
162        assert!(decode_file(&data).is_empty());
163    }

165    #[test]
166    fn absurd_record_size_stops_the_chunk() {

168        let size_at = FILE_HEADER_BLOCK + usize::try_from(CHUNK_RECORDS_OFFSET).unwrap() + 4;
169        data[size_at..size_at + 4].copy_from_slice(&10u32.to_le_bytes()); // size < header
170        assert!(decode_file(&data).is_empty());

173    #[test]
174    fn record_extending_past_chunk_stops_the_chunk() {
175        // A valid-looking size (>=24, <=CHUNK_SIZE) that nonetheless runs past the
176        // end of the chunk must stop iteration, not read out of bounds.
177        let mut data = synthetic_evtx();
178        let size_at = FILE_HEADER_BLOCK + usize::try_from(CHUNK_RECORDS_OFFSET).unwrap() + 4;
179        data[size_at..size_at + 4].copy_from_slice(&0xFF00u32.to_le_bytes());
180        let fso = FILE_HEADER_BLOCK + 48; // free_space_offset → admit the header read
181        data[fso..fso + 4].copy_from_slice(&u32::try_from(CHUNK_SIZE).unwrap().to_le_bytes());
182        assert!(decode_file(&data).is_empty());
183    }
