1//! BinXml value-type decoding (scalar types).
2//!
3//! Each substitution value / inline value carries a 1-byte type id. This module
4//! decodes the scalar types into [`BinXmlValue`] and renders them to their
5//! canonical string form. Array variants (`0x80 | base`) and embedded BinXml
6//! (`0x21`) are handled by the deserializer (they expand / recurse in element
7//! context) and are reported here as [`ValueError::Unsupported`].
8//!
9//! Byte semantics follow the omerbenamram `evtx` crate (Apache/MIT) and libevtx:
10//! `String` is either *sized* (descriptor byte length, no prefix) or
11//! *len-prefixed* (`u16` char count); `Bool` is 4 bytes (nonzero = true);
12//! `SizeT` renders as hex; `Sid` is `8 + sub_count*4` bytes; FILETIME/SYSTEMTIME
13//! render as UTC ISO-8601.
14

20/// Error decoding a BinXml value.
21#[derive(Debug, Error, PartialEq, Eq, Clone)]
22pub enum ValueError {
23    /// A bounds-checked cursor read failed.
24    #[error("cursor: {0}")]
25    Cursor(#[from] CursorError),
26    /// The type id is not a known BinXml value type.
27    #[error("unknown value type {type_id:#04x}")]
28    UnknownType { type_id: u8 },
29    /// The type is known but handled elsewhere (arrays, embedded BinXml).
30    #[error("value type {type_id:#04x} is not handled by read_value (array/binxml)")]
31    Unsupported { type_id: u8 },
32    /// A SID sub-authority count exceeded the sane cap.
33    #[error("invalid SID: sub-authority count {count} exceeds cap")]
34    InvalidSid { count: u8 },
35    /// A FILETIME/SYSTEMTIME could not be represented.
36    #[error("invalid timestamp")]
37    InvalidTimestamp,
38}

71/// A decoded scalar BinXml value. Complex types (GUID/SID/timestamps) carry
72/// their already-rendered canonical string.
73#[derive(Debug, Clone, PartialEq)]
74pub enum BinXmlValue {
75    Null,
76    String(String),
77    AnsiString(String),
78    Int8(i8),
79    UInt8(u8),
80    Int16(i16),
81    UInt16(u16),
82    Int32(i32),
83    UInt32(u32),
84    Int64(i64),
85    UInt64(u64),
86    Real32(f32),
87    Real64(f64),
88    Bool(bool),
89    Binary(Vec<u8>),
90    Guid(String),
91    SizeT(String),
92    FileTime(String),
93    SysTime(String),
94    Sid(String),
95    HexInt32(u32),
96    HexInt64(u64),
97}
98
99impl BinXmlValue {
100    /// Render the value to its canonical UTF-8 string (the text that appears in
101    /// the decoded XML/JSON). `Null` renders as the empty string.
102    #[must_use]
103    pub fn render(&self) -> String {
104        match self {
105            Self::Null => String::new(),
106            Self::String(s)
107            | Self::AnsiString(s)
108            | Self::Guid(s)
109            | Self::SizeT(s)
110            | Self::FileTime(s)
111            | Self::SysTime(s)
112            | Self::Sid(s) => s.clone(),
113            Self::Int8(v) => v.to_string(),
114            Self::UInt8(v) => v.to_string(),
115            Self::Int16(v) => v.to_string(),
116            Self::UInt16(v) => v.to_string(),
117            Self::Int32(v) => v.to_string(),
118            Self::UInt32(v) => v.to_string(),
119            Self::Int64(v) => v.to_string(),
120            Self::UInt64(v) => v.to_string(),
121            Self::Real32(v) => v.to_string(),
122            Self::Real64(v) => v.to_string(),
123            Self::Bool(b) => b.to_string(),
124            Self::Binary(b) => to_hex(b),
125            Self::HexInt32(v) => format!("0x{v:08x}"),
126            Self::HexInt64(v) => format!("0x{v:016x}"),
127        }
128    }
129}

131/// Decode a value of `type_id` from `cur`. `size` is the descriptor byte length
132/// when the value is in a sized context (template substitution); `None` for
133/// inline self-describing values.
134pub fn read_value(
135    cur: &mut Cursor<'_>,
136    type_id: u8,
137    size: Option<usize>,
138) -> Result<BinXmlValue, ValueError> {
139    // Array variants expand the containing element — the deserializer's job.
140    if type_id & VT_ARRAY_FLAG != 0 {
141        return Err(ValueError::Unsupported { type_id });
142    }
143    match type_id {
144        VT_NULL => Ok(BinXmlValue::Null),
145        VT_STRING => {
146            let s = match size {
147                Some(bytes) => cur.read_utf16le_chars(bytes / 2)?,
148                None => cur.read_utf16le_len_prefixed()?,
149            };
150            Ok(BinXmlValue::String(s))
151        }
152        VT_ANSI_STRING => {
153            let n = match size {
154                Some(n) => n,
155                None => usize::from(cur.read_u16_le()?),
156            };
157            Ok(BinXmlValue::AnsiString(ansi_to_string(cur.take(n)?)))
158        }
159        VT_INT8 => Ok(BinXmlValue::Int8(cur.read_i8()?)),
160        VT_UINT8 => Ok(BinXmlValue::UInt8(cur.read_u8()?)),
161        VT_INT16 => Ok(BinXmlValue::Int16(cur.read_i16_le()?)),
162        VT_UINT16 => Ok(BinXmlValue::UInt16(cur.read_u16_le()?)),
163        VT_INT32 => Ok(BinXmlValue::Int32(cur.read_i32_le()?)),
164        VT_UINT32 => Ok(BinXmlValue::UInt32(cur.read_u32_le()?)),
165        VT_INT64 => Ok(BinXmlValue::Int64(cur.read_i64_le()?)),
166        VT_UINT64 => Ok(BinXmlValue::UInt64(cur.read_u64_le()?)),
167        VT_REAL32 => Ok(BinXmlValue::Real32(cur.read_f32_le()?)),
168        VT_REAL64 => Ok(BinXmlValue::Real64(cur.read_f64_le()?)),
169        VT_BOOL => Ok(BinXmlValue::Bool(cur.read_u32_le()? != 0)),
170        VT_BINARY => {
171            let n = match size {
172                Some(n) => n,
173                None => usize::from(cur.read_u16_le()?),
174            };
175            Ok(BinXmlValue::Binary(cur.take(n)?.to_vec()))
176        }
177        VT_GUID => Ok(BinXmlValue::Guid(read_guid(cur)?)),
178        VT_SIZET => match size {
179            Some(4) => Ok(BinXmlValue::SizeT(format!("0x{:08x}", cur.read_u32_le()?))),
180            _ => Ok(BinXmlValue::SizeT(format!("0x{:016x}", cur.read_u64_le()?))),
181        },
182        VT_FILETIME => Ok(BinXmlValue::FileTime(render_filetime(cur.read_u64_le()?)?)),
183        VT_SYSTIME => Ok(BinXmlValue::SysTime(read_systime(cur)?)),
184        VT_SID => Ok(BinXmlValue::Sid(read_sid(cur)?)),
185        VT_HEX32 => Ok(BinXmlValue::HexInt32(cur.read_u32_le()?)),
186        VT_HEX64 => Ok(BinXmlValue::HexInt64(cur.read_u64_le()?)),
187        // Embedded BinXml recurses in element context — handled by the deserializer.
188        VT_BINXML => Err(ValueError::Unsupported { type_id }),
189        _ => Err(ValueError::UnknownType { type_id }),
190    }
191}

193/// Lower-case hex of a byte slice, no separator.
194fn to_hex(bytes: &[u8]) -> String {
195    use std::fmt::Write;
196    let mut s = String::with_capacity(bytes.len().saturating_mul(2));
197    for b in bytes {
198        let _ = write!(s, "{b:02x}");
199    }
200    s
201}

203/// Decode an ANSI (latin-1) string, dropping NUL bytes.
204fn ansi_to_string(raw: &[u8]) -> String {
205    raw.iter().filter(|&&b| b != 0).map(|&b| b as char).collect()
206}

208/// Read a 16-byte GUID and render it canonically (`xxxxxxxx-xxxx-…`, lower-case).
209fn read_guid(cur: &mut Cursor<'_>) -> Result<String, ValueError> {
210    let d1 = cur.read_u32_le()?;
211    let d2 = cur.read_u16_le()?;
212    let d3 = cur.read_u16_le()?;
213    let d4 = cur.take(8)?;
214    let (clock, node) = d4.split_at(2);
215    Ok(format!(
216        "{d1:08x}-{d2:04x}-{d3:04x}-{}-{}",
217        to_hex(clock),
218        to_hex(node)
219    ))
220}

222/// Read a SID (`revision u8, sub_count u8, authority 48-bit BE, sub×u32 LE`) and
223/// render `S-…`. Rejects an absurd sub-authority count before allocating.
224fn read_sid(cur: &mut Cursor<'_>) -> Result<String, ValueError> {
225    use std::fmt::Write;
226    let revision = cur.read_u8()?;
227    let sub_count = cur.read_u8()?;
228    if sub_count > MAX_SID_SUBAUTHORITIES {
229        return Err(ValueError::InvalidSid { count: sub_count });
230    }
231    let authority = cur
232        .take(6)?
233        .iter()
234        .fold(0u64, |acc, &b| (acc << 8) | u64::from(b));
235    let mut s = format!("S-{revision}-{authority}");
236    for _ in 0..sub_count {
237        let sub = cur.read_u32_le()?;
238        let _ = write!(s, "-{sub}");
239    }
240    Ok(s)
241}

243/// Convert a Windows FILETIME (100ns since 1601-01-01 UTC) to UTC ISO-8601.
244fn render_filetime(filetime: u64) -> Result<String, ValueError> {
245    /// Seconds between 1601-01-01 and the Unix epoch.
246    const EPOCH_DIFF_SECS: i64 = 11_644_473_600;
247    let secs_1601 = i64::try_from(filetime / 10_000_000).unwrap_or(i64::MAX);
248    let nanos = u32::try_from((filetime % 10_000_000) * 100).unwrap_or(0);
249    let unix_secs = secs_1601.saturating_sub(EPOCH_DIFF_SECS);
250    let dt = chrono::DateTime::from_timestamp(unix_secs, nanos)
251        .ok_or(ValueError::InvalidTimestamp)?;
252    Ok(dt.format("%Y-%m-%dT%H:%M:%S%.6fZ").to_string())
253}
254
255/// Read a 16-byte SYSTEMTIME and render as UTC ISO-8601.
256fn read_systime(cur: &mut Cursor<'_>) -> Result<String, ValueError> {
257    let year = cur.read_u16_le()?;
258    let month = cur.read_u16_le()?;
259    let _day_of_week = cur.read_u16_le()?;
260    let day = cur.read_u16_le()?;
261    let hour = cur.read_u16_le()?;
262    let minute = cur.read_u16_le()?;
263    let second = cur.read_u16_le()?;
264    let milli = cur.read_u16_le()?;
265    let date = chrono::NaiveDate::from_ymd_opt(i32::from(year), u32::from(month), u32::from(day))
266        .ok_or(ValueError::InvalidTimestamp)?;
267    let dt = date
268        .and_hms_milli_opt(
269            u32::from(hour),
270            u32::from(minute),
271            u32::from(second),
272            u32::from(milli),
273        )
274        .ok_or(ValueError::InvalidTimestamp)?;
275    Ok(dt.format("%Y-%m-%dT%H:%M:%S%.6fZ").to_string())
276}
