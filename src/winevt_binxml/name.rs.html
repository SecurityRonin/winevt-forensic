1//! Per-chunk BinXml name resolution (the element/attribute string table).
2//!
3//! A name reference in the token stream is a `u32` **chunk-relative** offset. If
4//! that offset equals the stream position right after it, the name is defined
5//! *inline* (read it here, advance the cursor); otherwise it is a back-reference
6//! to a name defined earlier in the chunk.
7//!
8//! Each name structure is self-describing at its offset —
9//! `next_string u32, hash u16, char_count u16, UTF-16LE chars, NUL u16` — so
10//! resolution is **lazy and per-offset**. We never follow the `next_string`
11//! hash-bucket chain, which makes a hostile linked-list cycle structurally
12//! impossible (the reference walks those chains; we do not).
13
14#![allow(clippy::doc_markdown)] // "BinXml" appears throughout these docs

16use std::collections::HashMap;
17
18use crate::cursor::{Cursor, CursorError};
19use thiserror::Error;

21/// Error resolving a BinXml name.
22#[derive(Debug, Error, PartialEq, Eq, Clone)]
23pub enum NameError {
24    /// A bounds-checked cursor read failed (truncated / out-of-range offset).
25    #[error("cursor: {0}")]
26    Cursor(#[from] CursorError),
27}
28
29/// Cache of resolved names keyed by chunk-relative offset.
30#[derive(Debug, Default)]
31pub struct NameCache {
32    map: HashMap<usize, String>,
33}
34
35impl NameCache {
36    /// Create an empty cache.
37    #[must_use]
38    pub fn new() -> Self {
39        Self::default()
40    }
41
42    /// Number of distinct names resolved so far.
43    #[must_use]
44    pub fn len(&self) -> usize {
45        self.map.len()
46    }
47
48    /// Whether no names have been resolved yet.
49    #[must_use]
50    pub fn is_empty(&self) -> bool {
51        self.map.is_empty()
52    }
53
54    /// Read a name reference at `cur` and return the name. `chunk` is the full
55    /// chunk slice `cur` is reading (the addressing base for back-references).
56    pub fn read_name_ref(
57        &mut self,
58        cur: &mut Cursor<'_>,
59        chunk: &[u8],
60    ) -> Result<String, NameError> {
61        let name_offset = cur.read_u32_le()? as usize;
62        let pos_after = cur.position();
63        if name_offset == pos_after {
64            // Inline definition: the struct begins here — read from the main
65            // cursor so it advances past the whole inline name.
66            let name = read_name_fields(cur)?;
67            cur.skip(2)?; // NUL terminator
68            self.map.insert(name_offset, name.clone());
69            Ok(name)
70        } else if let Some(cached) = self.map.get(&name_offset) {
71            Ok(cached.clone())
72        } else {
73            // Back-reference: read the self-describing struct at the offset via a
74            // throwaway sub-cursor (the main cursor is not advanced into it).
75            let mut sub = Cursor::at(chunk, name_offset);
76            let name = read_name_fields(&mut sub)?;
77            self.map.insert(name_offset, name.clone());
78            Ok(name)
79        }
80    }
81}
82
83/// Read `next_string u32, hash u16, char_count u16, UTF-16LE chars` and return
84/// the decoded name. Does not consume the trailing NUL terminator.
85fn read_name_fields(cur: &mut Cursor<'_>) -> Result<String, NameError> {
86    cur.skip(4)?; // next_string
87    cur.skip(2)?; // hash
88    let count = cur.read_u16_le()?;
89    Ok(cur.read_utf16le_chars(usize::from(count))?)
90}

92#[cfg(test)]

96    /// `next=0, hash=0, char_count, "<s>", NUL` — a name struct body.
97    fn name_struct(s: &str) -> Vec<u8> {
98        let units: Vec<u16> = s.encode_utf16().collect();
99        let mut v = Vec::new();
100        v.extend_from_slice(&0u32.to_le_bytes()); // next_string
101        v.extend_from_slice(&0u16.to_le_bytes()); // hash
102        v.extend_from_slice(&(units.len() as u16).to_le_bytes()); // char_count
103        for u in &units {
104            v.extend_from_slice(&u.to_le_bytes());
105        }
106        v.extend_from_slice(&0u16.to_le_bytes()); // NUL
107        v
108    }

110    #[test]
111    fn inline_name_is_read_and_cursor_advances() {
112        // offset 0: name_offset = 4 (points right after the u32 → inline)
113        // offset 4: name struct "Hi"
114        let mut chunk = Vec::new();
115        chunk.extend_from_slice(&4u32.to_le_bytes());
116        chunk.extend_from_slice(&name_struct("Hi"));
117
118        let mut cache = NameCache::new();
119        let mut cur = Cursor::new(&chunk);
120        let name = cache.read_name_ref(&mut cur, &chunk).unwrap();
121        assert_eq!(name, "Hi");
122        // cursor must end past the whole inline struct (4 offset + 10 + 2*2 = 18)
123        assert_eq!(cur.position(), 18);
124        assert_eq!(cache.len(), 1);
125    }
126
127    #[test]
128    fn back_reference_resolves_from_offset() {
129        // offset 0: a name struct "Channel" at 0
130        // then a ref u32 = 0 read from a later position
131        let mut chunk = name_struct("Channel");
132        let ref_pos = chunk.len();
133        chunk.extend_from_slice(&0u32.to_le_bytes()); // name_offset = 0 (back-ref)
134
135        let mut cache = NameCache::new();
136        let mut cur = Cursor::new(&chunk);
137        cur.seek(ref_pos).unwrap();
138        let name = cache.read_name_ref(&mut cur, &chunk).unwrap();
139        assert_eq!(name, "Channel");
140        // back-ref does not advance into the struct; only the u32 was consumed
141        assert_eq!(cur.position(), ref_pos + 4);
142    }

144    #[test]
145    fn back_reference_uses_cache_on_second_hit() {
146        let mut chunk = name_struct("System");
147        let p1 = chunk.len();
148        chunk.extend_from_slice(&0u32.to_le_bytes());
149        let p2 = chunk.len();
150        chunk.extend_from_slice(&0u32.to_le_bytes());
151
152        let mut cache = NameCache::new();
153        let mut cur = Cursor::new(&chunk);
154        cur.seek(p1).unwrap();
155        assert_eq!(cache.read_name_ref(&mut cur, &chunk).unwrap(), "System");
156        cur.seek(p2).unwrap();
157        assert_eq!(cache.read_name_ref(&mut cur, &chunk).unwrap(), "System");
158        assert_eq!(cache.len(), 1, "same offset resolved once, then cached");
159    }

161    #[test]
162    fn out_of_range_offset_is_error_not_panic() {
163        // name_offset points way past the chunk → Err
164        let chunk = 9999u32.to_le_bytes().to_vec();
165        let mut cache = NameCache::new();
166        let mut cur = Cursor::new(&chunk);
167        assert!(cache.read_name_ref(&mut cur, &chunk).is_err());
168    }
