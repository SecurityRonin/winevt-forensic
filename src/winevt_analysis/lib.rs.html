4//! and return a `Vec<EvtxDetection>`.  They are pure functions with no I/O.
5//!
6//! Call [`detect_all`] to run every detector and get results sorted by
7//! timestamp then MITRE technique ID.

82/// Category of a Windows Event Log detection.
83#[derive(Debug, Clone, Copy, PartialEq, Eq, serde::Serialize)]
84pub enum EvtxDetectionKind {
85    /// A known-vulnerable driver service was installed (T1068 — Exploit Public-Facing Application / BYOVD).
86    ByovdDriverInstall,
87    /// A security-sensitive registry value controlling HVCI or VBS was modified (T1562.001).
88    HvciRegistryTamper,
89    /// Shadow copies were deleted — inhibit system recovery (T1490).
90    VssDeletion,
91    /// Hyper-V VM(s) shut down — pre-encryption step in QWCrypt-style ransomware (T1486).
92    HypervVmShutdown,
93    /// A scheduled task was created — persistence mechanism (T1053.005).
94    ScheduledTaskCreation,
95    /// A known QWCrypt/RedCurl binary executed (T1486).
96    QwcryptProcessExecution,
97    /// A PowerShell script block matched QWCrypt/RedCurl-specific patterns (T1059.001).
98    PsScriptBlockQwcrypt,
99    /// A LOLBin process initiated a WebDAV connection — staging or exfiltration (T1102).
100    WebdavLolbinUsage,
101    /// A ransom note file was created on disk (T1486 — Data Encrypted for Impact).
102    RansomNoteCreated,
103    /// WebClient (Mini-Redirector) service started — precursor to WebDAV payload delivery (T1102).
104    WebClientServiceStart,
105    /// Sysinternals AD Explorer first-run registry key created — domain recon tombstone (T1087).
106    AdExplorerRecon,
107    /// Zemana signed driver loaded from a non-standard path — BYOVD EDR killer (T1068).
108    ZemanaDriverLoad,
109    /// srvcli.dll or netutils.dll loaded from a user-writable path — DLL sideload (T1574.002).
110    DllSideloadHijack,
111    /// ConsoleHost_history.txt deleted — PowerShell history anti-forensics (T1070.003).
112    PsHistoryWipe,
113    /// RPivot or Chisel tunnel tool executed — reverse proxy / SOCKS5 (T1090).
114    RpivotChisel,
115    /// WMI permanent event consumer created or Impacket wmiexec output-redirect detected (T1047).
116    WmiLateralMovement,
117    /// 7-Zip executed with header-encryption flag (-mhe) from a suspicious parent (T1560.001).
118    SevenZipStagingEncrypted,
119    /// Scheduled task with browser-update name but action path outside Program Files (T1053.005).
120    FakeBrowserTask,
121    /// Non-browser process issued a DNS query for a .workers.dev domain — Cloudflare C2 (T1102).
122    WorkersDevDnsQuery,
123    /// vssadmin.exe or wmic.exe deleted VSS shadow copies — inhibit system recovery (T1490).
124    VssAdminWmicDelete,
125    /// Cluster of ≥5 taskkill /IM targeting canonical AV/SQL processes within 60s (T1562.001).
126    TaskkillAvCluster,
127    /// Cluster of ≥3 service stops targeting canonical AV/backup services within 60s (T1489).
128    ServiceStopAvSet,
129    /// bcdedit.exe modified boot recovery options — pre-encryption staging (T1490).
130    BcdeditRecoveryTamper,
131    /// Event logs cleared via wevtutil.exe or PowerShell Clear-EventLog (T1070.001).
132    WevtutilLogClear,
133    /// Windows Defender or AV disabled via PowerShell Set-MpPreference or EID 5001 (T1562.001).
134    DefenderDisabled,
135    /// LSASS credential dump via comsvcs.dll MiniDump or Sysmon process-access (T1003.001).
136    ComsvcslsassDump,
137    /// RMM tool binary dropped outside standard Program Files paths (T1219).
138    RmmToolInstall,
139    /// fDenyTSConnections set to 0 — RDP enabled via registry (T1021.001).
140    RdpEnabled,
141    /// New local user account created (EID 4720) or added to local Admins group (EID 4732) (T1136.001).
142    LocalAdminCreation,
143    /// Remote access to Windows admin share (ADMIN$, C$, IPC$) from a non-local IP (T1021.002).
144    SmbAdminShareAccess,
145    /// LOLBin process spawned directly by explorer.exe — user executed a malicious LNK/script (T1204.002).
146    ExplorerLolbinExecution,
147}

149/// A single detection produced by an EVTX detector.
150#[derive(Debug, Clone, serde::Serialize)]
151pub struct EvtxDetection {
152    pub kind: EvtxDetectionKind,
153    /// MITRE ATT&CK technique ID.
154    pub mitre_technique_id: &'static str,
155    /// ATT&CK tactic name.
156    pub tactic: &'static str,
157    /// Human-readable finding description.
158    pub description: String,
159    /// Evidence items: field names and values that triggered this detection.
160    pub evidence: Vec<String>,
161    /// Nanosecond-precision timestamp of the triggering event.
162    pub timestamp_ns: i64,
163    /// Windows Event ID of the triggering event.
164    pub event_id: u32,
165    /// Event log channel the event was read from.
166    pub channel: String,
167}

169/// Run all EVTX detectors over a slice of events.
170///
171/// Results are sorted by `timestamp_ns` then `mitre_technique_id` for
172/// deterministic, diff-friendly output.
173pub fn detect_all(events: &[EvtxEvent]) -> Vec<EvtxDetection> {
174    let mut results = Vec::new();
175    results.extend(detect_byovd_driver_install(events));
176    results.extend(detect_hvci_registry_tamper(events));
177    results.extend(detect_vss_deletion(events));
178    results.extend(detect_hyperv_vm_shutdown(events));
179    results.extend(detect_scheduled_task_creation(events));
180    results.extend(detect_qwcrypt_process(events));
181    results.extend(detect_ps_qwcrypt_patterns(events));
182    results.extend(detect_webdav_lolbin(events));
183    results.extend(detect_ransom_note_creation(events));
184    results.extend(detect_webclient_service_start(events));
185    results.extend(detect_adexplorer_recon(events));
186    results.extend(detect_zemana_driver_load(events));
187    results.extend(detect_dll_sideload(events));
188    results.extend(detect_ps_history_wipe(events));
189    results.extend(detect_rpivot_chisel(events));
190    results.extend(detect_wmi_lateral_movement(events));
191    results.extend(detect_sevenz_staging(events));
192    results.extend(detect_fake_browser_task(events));
193    results.extend(detect_workers_dev_dns(events));
194    results.extend(detect_vssadmin_wmic(events));
195    results.extend(detect_taskkill_av_cluster(events));
196    results.extend(detect_service_stop_avset(events));
197    results.extend(detect_bcdedit_recovery(events));
198    results.extend(detect_wevtutil_cl(events));
199    results.extend(detect_defender_disable(events));
200    results.extend(detect_comsvcs_lsass(events));
201    results.extend(detect_rmm_install(events));
202    results.extend(detect_rdp_enable(events));
203    results.extend(detect_local_admin_creation(events));
204    results.extend(detect_smb_admin_share(events));
205    results.extend(detect_explorer_lolbin(events));
206    results.sort_by(|a, b| {
207        a.timestamp_ns
208            .cmp(&b.timestamp_ns)
209            .then(a.mitre_technique_id.cmp(b.mitre_technique_id))
210    });
211    results
212}

214#[cfg(test)]
215pub(crate) mod test_helpers {
216    use std::collections::HashMap;
217    use winevt_core::EvtxEvent;
218
219    pub fn make_event(event_id: u32, channel: &str, data: &[(&str, &str)]) -> EvtxEvent {
220        EvtxEvent {
221            event_id,
222            channel: channel.to_string(),
223            timestamp_ns: 1_700_000_000_000_000_000,
224            computer: "test-pc".to_string(),
225            user_sid: None,
226            logon_id: None,
227            process_id: None,
228            thread_id: None,
229            data: data
230                .iter()
231                .map(|(k, v)| (k.to_string(), v.to_string()))
232                .collect::<HashMap<_, _>>(),
233        }
234    }
235}

237#[cfg(test)]
238mod tests {
239    use super::*;
240    use test_helpers::make_event;
241
242    #[test]
243    fn detect_all_empty_returns_empty() {
244        assert!(detect_all(&[]).is_empty());
245    }
246
247    #[test]
248    fn detect_all_byovd_event_detected() {
249        let ev = make_event(7045, "System", &[("ServiceName", "zamguard64")]);
250        let hits = detect_all(&[ev]);
251        assert!(hits.iter().any(|h| h.kind == EvtxDetectionKind::ByovdDriverInstall));
252    }
253
254    #[test]
255    fn detect_all_results_sorted_by_timestamp_then_mitre() {
256        let ev1 = make_event(7045, "System", &[("ServiceName", "zamguard64")]);
257        let ev2 = make_event(8193, "Application", &[("Source", "VSS")]);
258        let hits = detect_all(&[ev1, ev2]);
259        for w in hits.windows(2) {
260            assert!(
261                w[0].timestamp_ns <= w[1].timestamp_ns,
262                "results must be sorted by timestamp"
263            );
264        }
265    }
