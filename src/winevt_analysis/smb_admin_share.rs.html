1//! Detect lateral movement via SMB administrative shares (T1021.002).
2
3use forensicnomicon::heuristics::evtx::{ADMIN_SHARE_NAMES, EID_SMB_SHARE_ACCESS};
4use winevt_core::EvtxEvent;
5
6use crate::{EvtxDetection, EvtxDetectionKind};
7
8/// Detect remote access to Windows administrative shares via Security EID 5140.
9///
10/// EID 5140 fires when a network share is accessed. A remote IP (not localhost)
11/// accessing `ADMIN$`, `C$`, `D$`, `IPC$`, etc. is a strong lateral movement
12/// signal (T1021.002 — Remote Services: SMB/Windows Admin Shares).
13///
14/// False positives: legitimate RPC/Named-pipe connections use `IPC$` — correlate
15/// with surrounding events (service install, psexec patterns) for high confidence.
16pub fn detect_smb_admin_share(events: &[EvtxEvent]) -> Vec<EvtxDetection> {
17    events
18        .iter()
19        .filter(|ev| ev.event_id == EID_SMB_SHARE_ACCESS && ev.channel == "Security")
20        .filter_map(|ev| {
21            let share_field = ev.data.get("ShareName").map(String::as_str).unwrap_or("");
22            let share = share_name_component(share_field);
23            let matched = ADMIN_SHARE_NAMES
24                .iter()
25                .find(|&&name| name.eq_ignore_ascii_case(share))?;
26            let ip = ev.data.get("IpAddress").map(String::as_str).unwrap_or("-");
27            if is_local_ip(ip) {
28                return None;
29            }
30            let user = ev
31                .data
32                .get("SubjectUserName")
33                .map(String::as_str)
34                .unwrap_or("unknown");
35            Some(EvtxDetection {
36                kind: EvtxDetectionKind::SmbAdminShareAccess,
37                mitre_technique_id: "T1021.002",
38                tactic: "Lateral Movement",
39                description: format!(
40                    "Remote SMB admin share access: '{matched}' from {ip} by '{user}'"
41                ),
42                evidence: vec![
43                    format!("ShareName={share_field}"),
44                    format!("IpAddress={ip}"),
45                    format!("SubjectUserName={user}"),
46                ],
47                timestamp_ns: ev.timestamp_ns,
48                event_id: ev.event_id,
49                channel: ev.channel.clone(),
50            })
51        })
52        .collect()
53}
54
55/// Extract the share name component from a ShareName field value.
56/// EID 5140 `ShareName` is formatted as `\\<computer>\<share>`.
57/// Returns the last path component in uppercase.
58fn share_name_component(share_field: &str) -> &str {
59    share_field
60        .rsplit('\\')
61        .find(|s| !s.is_empty())
62        .unwrap_or(share_field)
63}
64
65/// Returns true if the IP address string represents a local/loopback connection.
66fn is_local_ip(ip: &str) -> bool {
67    matches!(ip.trim(), "127.0.0.1" | "::1" | "-" | "")
68}

70#[cfg(test)]
71mod tests {
72    use super::*;
73    use crate::test_helpers::make_event;

75    fn share_access_event(share_name: &str, ip_address: &str) -> EvtxEvent {
76        make_event(
77            EID_SMB_SHARE_ACCESS,
78            "Security",
79            &[
80                ("SubjectUserName", "attacker"),
81                ("ShareName", share_name),
82                ("IpAddress", ip_address),
83            ],
84        )
85    }
86
87    #[test]
88    fn admin_dollar_from_remote_ip_detected() {
89        let ev = share_access_event("\\\\*\\ADMIN$", "10.0.0.42");
90        let hits = detect_smb_admin_share(&[ev]);
91        assert!(!hits.is_empty());
92        assert_eq!(hits[0].kind, EvtxDetectionKind::SmbAdminShareAccess);
93        assert_eq!(hits[0].mitre_technique_id, "T1021.002");
94    }
95
96    #[test]
97    fn c_dollar_from_remote_ip_detected() {
98        let ev = share_access_event("\\\\*\\C$", "192.168.1.100");
99        assert!(!detect_smb_admin_share(&[ev]).is_empty());
100    }
101
102    #[test]
103    fn ipc_dollar_from_remote_ip_detected() {
104        let ev = share_access_event("\\\\FILESERVER\\IPC$", "10.1.2.3");
105        assert!(!detect_smb_admin_share(&[ev]).is_empty());
106    }
107
108    #[test]
109    fn admin_dollar_from_localhost_not_detected() {
110        let ev = share_access_event("\\\\*\\ADMIN$", "127.0.0.1");
111        assert!(detect_smb_admin_share(&[ev]).is_empty());
112    }

139    #[test]
140    fn evidence_contains_share_name_and_ip() {
141        let ev = share_access_event("\\\\*\\C$", "172.16.0.50");
142        let hits = detect_smb_admin_share(&[ev]);
143        assert!(!hits.is_empty());
144        let combined = hits[0].evidence.join(" ");
145        assert!(combined.contains("C$") || combined.contains("ShareName"));
146        assert!(combined.contains("172.16.0.50") || combined.contains("IpAddress"));
