1//! Detect ransomware service-stop cluster targeting AV/backup services (T1489).

5    ransomware::{
6        RANSOMWARE_KILL_WINDOW_NS, RANSOMWARE_SERVICE_STOP_CLUSTER_THRESHOLD,
7        RANSOMWARE_STOP_SERVICES,
8    },
9};
10use winevt_core::EvtxEvent;

12use crate::{EvtxDetection, EvtxDetectionKind};

14/// Detect a cluster of `net stop <svc>` or `sc.exe stop <svc>` executions
15/// where ≥3 targets are in the canonical AV/backup service list within 60s.
16///
17/// Single service stops have medium FP risk.  A cluster of ≥3 canonical
18/// services stopped within `RANSOMWARE_KILL_WINDOW_NS` is near-zero-FP
19/// (T1489 — Service Stop; T1562.001 — Impair Defenses).
20pub fn detect_service_stop_avset(events: &[EvtxEvent]) -> Vec<EvtxDetection> {
21    // Collect matching (timestamp, svc_name, event) tuples
22    let mut matches: Vec<(i64, &str, &EvtxEvent)> = events
23        .iter()
24        .filter(|ev| is_process_event(ev))
25        .filter_map(|ev| {
26            stopped_canonical_service(ev).map(|svc| (ev.timestamp_ns, svc, ev))
27        })
28        .collect();
29
30    matches.sort_by_key(|(ts, _, _)| *ts);
31
32    let mut detections = Vec::new();
33    let mut i = 0;
34    while i < matches.len() {
35        let window_start = matches[i].0;
36        let window_end = window_start + RANSOMWARE_KILL_WINDOW_NS;
37        let window: Vec<_> = matches[i..]
38            .iter()
39            .take_while(|(ts, _, _)| *ts <= window_end)
40            .collect();
41        if window.len() >= RANSOMWARE_SERVICE_STOP_CLUSTER_THRESHOLD {
42            let services: Vec<String> = window.iter().map(|(_, s, _)| s.to_string()).collect();
43            let last_ev = window.last().unwrap().2;
44            detections.push(EvtxDetection {
45                kind: EvtxDetectionKind::ServiceStopAvSet,
46                mitre_technique_id: "T1489",
47                tactic: "Impact",
48                description: format!(
49                    "Ransomware AV/backup service-stop cluster: {} stops in {}ms — [{}]",
50                    window.len(),
51                    (window.last().unwrap().0 - window_start) / 1_000_000,
52                    services.join(", ")
53                ),
54                evidence: services
55                    .iter()
56                    .map(|s| format!("stopped={s}"))
57                    .collect(),
58                timestamp_ns: window_start,
59                event_id: last_ev.event_id,
60                channel: last_ev.channel.clone(),
61            });
62            i += window.len();
63        } else {
64            i += 1;
65        }
66    }
67    detections
68}

70fn is_process_event(ev: &EvtxEvent) -> bool {
71    (ev.event_id == EID_PROCESS_CREATE && ev.channel == "Security")
72        || (ev.event_id == EID_SYSMON_PROCESS_CREATE && ev.channel == SYSMON_CHANNEL)
73}
74
75fn basename(path: &str) -> &str {
76    path.rsplit(|c| c == '\\' || c == '/').next().unwrap_or(path)
77}
78
79fn stopped_canonical_service<'a>(ev: &EvtxEvent) -> Option<&'a str> {
80    let img = ev
81        .data
82        .get("Image")
83        .or_else(|| ev.data.get("NewProcessName"))
84        .map(String::as_str)
85        .unwrap_or("");
86    let base = basename(img).to_lowercase();
87    if base != "net.exe" && base != "net1.exe" && base != "sc.exe" {
88        return None;
89    }
90    let cl = ev.data.get("CommandLine").map(String::as_str).unwrap_or("");
91    let cl_lower = cl.to_lowercase();
92    if !cl_lower.contains("stop") {
93        return None;
94    }
95    RANSOMWARE_STOP_SERVICES
96        .iter()
97        .find(|&&svc| cl_lower.contains(svc))
98        .copied()
99}

101#[cfg(test)]
102mod tests {
103    use super::*;
104    use crate::test_helpers::make_event;
105
106    fn stop_event(svc: &str, ts: i64) -> EvtxEvent {
107        let mut ev = make_event(
108            EID_SYSMON_PROCESS_CREATE,
109            SYSMON_CHANNEL,
110            &[
111                ("Image", "C:\\Windows\\System32\\net.exe"),
112                ("CommandLine", &format!("net stop {svc}")),
113            ],
114        );
115        ev.timestamp_ns = ts;
116        ev
117    }
118
119    const BASE_TS: i64 = 1_700_000_000_000_000_000;
120    const SEC: i64 = 1_000_000_000;
121
122    #[test]
123    fn cluster_of_three_detected() {
124        let svcs = ["veeambackupsvc", "gxvss", "sqlserveragent"];
125        let events: Vec<_> = svcs
126            .iter()
127            .enumerate()
128            .map(|(i, s)| stop_event(s, BASE_TS + (i as i64) * 5 * SEC))
129            .collect();
130        let hits = detect_service_stop_avset(&events);
131        assert!(!hits.is_empty());
132        assert_eq!(hits[0].kind, EvtxDetectionKind::ServiceStopAvSet);
133        assert_eq!(hits[0].mitre_technique_id, "T1489");
134    }
135
136    #[test]
137    fn cluster_of_two_not_detected() {
138        let svcs = ["veeambackupsvc", "gxvss"];
139        let events: Vec<_> = svcs
140            .iter()
141            .enumerate()
142            .map(|(i, s)| stop_event(s, BASE_TS + (i as i64) * SEC))
143            .collect();
144        assert!(detect_service_stop_avset(&events).is_empty());
145    }
146
147    #[test]
148    fn three_stops_outside_window_not_detected() {
149        let svcs = ["veeambackupsvc", "gxvss", "sqlserveragent"];
150        // 25s apart → total span 50s — wait, threshold is 60s and 25*2=50s, so still in window
151        // Use 35s apart → total span 70s > 60s window
152        let events: Vec<_> = svcs
153            .iter()
154            .enumerate()
155            .map(|(i, s)| stop_event(s, BASE_TS + (i as i64) * 35 * SEC))
156            .collect();
157        assert!(detect_service_stop_avset(&events).is_empty());
158    }

176            .enumerate()
