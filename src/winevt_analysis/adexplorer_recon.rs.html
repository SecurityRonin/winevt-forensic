1//! Detect Sysinternals AD Explorer first-run registry tombstone (T1087).
2
3use forensicnomicon::heuristics::evtx::{
4    ADEXPLORER_EULAACCEPTED_KEY_FRAGMENT, EID_SYSMON_REGISTRY_ADD, EID_SYSMON_REGISTRY_MODIFY,
5    SYSMON_CHANNEL,
6};
7use winevt_core::EvtxEvent;
8
9use crate::{EvtxDetection, EvtxDetectionKind};
10
11/// Detect Sysinternals AD Explorer domain recon via its EulaAccepted registry key.
12///
13/// AD Explorer (ADExplorer.exe / ADExplorer64.exe) writes
14/// `HKCU\Software\Sysinternals\Active Directory Explorer\EulaAccepted=1` on
15/// first run.  This key persists even after the binary is deleted, making it a
16/// durable forensic tombstone (T1087).  Sysmon EID 12 (RegistryEvent object
17/// create) or EID 13 (RegistryEvent value set) with `TargetObject` containing
18/// the key path fragment fires on first run.
19pub fn detect_adexplorer_recon(events: &[EvtxEvent]) -> Vec<EvtxDetection> {
20    events
21        .iter()
22        .filter(|ev| {
23            (ev.event_id == EID_SYSMON_REGISTRY_ADD || ev.event_id == EID_SYSMON_REGISTRY_MODIFY)
24                && ev.channel == SYSMON_CHANNEL
25        })
26        .filter_map(|ev| {
27            let key = ev.data.get("TargetObject")?;
28            if !key.contains(ADEXPLORER_EULAACCEPTED_KEY_FRAGMENT) {
29                return None;
30            }
31            Some(EvtxDetection {
32                kind: EvtxDetectionKind::AdExplorerRecon,
33                mitre_technique_id: "T1087",
34                tactic: "Discovery",
35                description: format!(
36                    "Sysinternals AD Explorer first-run registry key written — domain recon tombstone: '{key}'"
37                ),
38                evidence: vec![format!("TargetObject={key}")],
39                timestamp_ns: ev.timestamp_ns,
40                event_id: ev.event_id,
41                channel: ev.channel.clone(),
42            })
43        })
44        .collect()
45}

47#[cfg(test)]
48mod tests {
49    use super::*;
50    use crate::test_helpers::make_event;
51
52    fn adexplorer_reg_create() -> EvtxEvent {
53        make_event(
54            EID_SYSMON_REGISTRY_ADD,
55            SYSMON_CHANNEL,
56            &[(
57                "TargetObject",
58                "HKCU\\Software\\Sysinternals\\Active Directory Explorer\\EulaAccepted",
59            )],
60        )
61    }
62
63    #[test]
64    fn adexplorer_registry_create_detected() {
65        let ev = adexplorer_reg_create();
66        let hits = detect_adexplorer_recon(&[ev]);
67        assert!(!hits.is_empty());
68        assert_eq!(hits[0].kind, EvtxDetectionKind::AdExplorerRecon);
69        assert_eq!(hits[0].mitre_technique_id, "T1087");
70    }
71
72    #[test]
73    fn adexplorer_registry_modify_detected() {
74        let ev = make_event(
75            EID_SYSMON_REGISTRY_MODIFY,
76            SYSMON_CHANNEL,
77            &[(
78                "TargetObject",
79                "HKCU\\Software\\Sysinternals\\Active Directory Explorer\\EulaAccepted",
80            )],
81        );
82        assert!(!detect_adexplorer_recon(&[ev]).is_empty());
83    }
84
85    #[test]
86    fn unrelated_registry_key_not_detected() {
87        let ev = make_event(
88            EID_SYSMON_REGISTRY_ADD,
89            SYSMON_CHANNEL,
90            &[("TargetObject", "HKCU\\Software\\Microsoft\\Office\\16.0\\Common")],
91        );
92        assert!(detect_adexplorer_recon(&[ev]).is_empty());
93    }
94
95    #[test]
96    fn wrong_channel_not_detected() {
97        let ev = make_event(
98            EID_SYSMON_REGISTRY_ADD,
99            "Security",
100            &[(
101                "TargetObject",
102                "HKCU\\Software\\Sysinternals\\Active Directory Explorer\\EulaAccepted",
103            )],
104        );
105        assert!(detect_adexplorer_recon(&[ev]).is_empty());
106    }
107
108    #[test]
109    fn evidence_contains_registry_key() {
110        let ev = adexplorer_reg_create();
111        let hits = detect_adexplorer_recon(&[ev]);
112        assert!(!hits.is_empty());
113        let combined = hits[0].evidence.join(" ");
114        assert!(combined.contains("Active Directory Explorer"));
115    }
116
117    #[test]
118    fn tactic_is_discovery() {
119        let ev = adexplorer_reg_create();
120        let hits = detect_adexplorer_recon(&[ev]);
121        assert!(!hits.is_empty());
122        assert_eq!(hits[0].tactic, "Discovery");
123    }
