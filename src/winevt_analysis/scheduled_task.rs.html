1//! Detect scheduled task creation events (T1053.005).
2
3use forensicnomicon::heuristics::evtx::EID_SECURITY_TASK_CREATED;
4use winevt_core::EvtxEvent;
5
6use crate::{EvtxDetection, EvtxDetectionKind};
7
8/// TaskScheduler/Operational EID 106 — task registered.
9const EID_TASKSCHEDULER_TASK_REGISTERED: u32 = 106;
10
11/// Detect scheduled task creation.
12///
13/// Fires on Security EID 4698 (task created, from Object Access audit) and
14/// TaskScheduler/Operational EID 106 (task registered).  QWCrypt/RedCurl
15/// installs a scheduled task for persistence after the initial DLL sideload
16/// (T1053.005).
17///
18/// Returns one detection per matching event.
19pub fn detect_scheduled_task_creation(events: &[EvtxEvent]) -> Vec<EvtxDetection> {
20    events
21        .iter()
22        .filter(|ev| {
23            (ev.event_id == EID_SECURITY_TASK_CREATED && ev.channel == "Security")
24                || (ev.event_id == EID_TASKSCHEDULER_TASK_REGISTERED
25                    && ev.channel.contains("TaskScheduler"))
26        })
27        .map(|ev| {
28            let task_name = ev
29                .data
30                .get("TaskName")
31                .map(String::as_str)
32                .unwrap_or("<unknown>");
33            EvtxDetection {
34                kind: EvtxDetectionKind::ScheduledTaskCreation,
35                mitre_technique_id: "T1053.005",
36                tactic: "Persistence",
37                description: format!("Scheduled task created: '{task_name}'"),
38                evidence: vec![format!("TaskName={task_name}"), format!("event_id={}", ev.event_id)],
39                timestamp_ns: ev.timestamp_ns,
40                event_id: ev.event_id,
41                channel: ev.channel.clone(),
42            }
43        })
44        .collect()
45}

47#[cfg(test)]
48mod tests {
49    use super::*;
50    use crate::test_helpers::make_event;
51
52    #[test]
53    fn security_4698_task_created_detected() {
54        let ev = make_event(
55            EID_SECURITY_TASK_CREATED,
56            "Security",
57            &[("TaskName", "\\Microsoft\\Windows\\EvilTask")],
58        );
59        let hits = detect_scheduled_task_creation(&[ev]);
60        assert!(!hits.is_empty());
61        assert_eq!(hits[0].kind, EvtxDetectionKind::ScheduledTaskCreation);
62        assert_eq!(hits[0].mitre_technique_id, "T1053.005");
63    }
64
65    #[test]
66    fn taskscheduler_106_detected() {
67        let ev = make_event(
68            EID_TASKSCHEDULER_TASK_REGISTERED,
69            "Microsoft-Windows-TaskScheduler/Operational",
70            &[("TaskName", "\\MyTask")],
71        );
72        assert!(!detect_scheduled_task_creation(&[ev]).is_empty());
73    }
74
75    #[test]
76    fn unrelated_event_not_detected() {
77        let ev = make_event(4624, "Security", &[("TaskName", "\\SomeTask")]);
78        assert!(detect_scheduled_task_creation(&[ev]).is_empty());
79    }
80
81    #[test]
82    fn evidence_contains_task_name() {
83        let ev = make_event(
84            EID_SECURITY_TASK_CREATED,
85            "Security",
86            &[("TaskName", "\\RedCurlPersist")],
87        );
88        let hits = detect_scheduled_task_creation(&[ev]);
89        assert!(!hits.is_empty());
90        let combined = [hits[0].description.as_str(), &hits[0].evidence.join(" ")].join(" ");
91        assert!(combined.contains("RedCurlPersist"));
92    }
