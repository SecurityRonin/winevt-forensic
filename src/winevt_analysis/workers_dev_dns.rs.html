1//! Detect non-browser DNS queries to .workers.dev — Cloudflare C2 (T1102).
2
3use forensicnomicon::heuristics::evtx::{
4    BROWSER_PROCESS_NAMES, CLOUDFLARE_WORKERS_DOMAIN_SUFFIX, EID_SYSMON_DNS_QUERY, SYSMON_CHANNEL,
5};
6use winevt_core::EvtxEvent;
7
8use crate::{EvtxDetection, EvtxDetectionKind};
9
10/// Detect non-browser processes querying `.workers.dev` domains.
11///
12/// QWCrypt/RedCurl uses Cloudflare Workers as rotating C2 redirectors (T1102).
13/// Sysmon EID 22 (DnsQuery) captures `QueryName` and `Image` (the process).
14/// Browsers legitimately query `.workers.dev`; every other process doing so is
15/// suspicious — especially `rundll32.exe`, `powershell.exe`, `pcalua.exe`,
16/// `python.exe`, and `ADNotificationManager.exe`.
17pub fn detect_workers_dev_dns(events: &[EvtxEvent]) -> Vec<EvtxDetection> {
18    events
19        .iter()
20        .filter(|ev| ev.event_id == EID_SYSMON_DNS_QUERY && ev.channel == SYSMON_CHANNEL)
21        .filter_map(|ev| {
22            let query = ev.data.get("QueryName")?;
23            if !query.ends_with(CLOUDFLARE_WORKERS_DOMAIN_SUFFIX) {
24                return None;
25            }
26            let image = ev.data.get("Image").map(String::as_str).unwrap_or("");
27            if is_browser(image) {
28                return None;
29            }
30            Some(EvtxDetection {
31                kind: EvtxDetectionKind::WorkersDevDnsQuery,
32                mitre_technique_id: "T1102",
33                tactic: "Command and Control",
34                description: format!(
35                    "Non-browser process queried Cloudflare Workers domain: '{query}' (process: '{image}')"
36                ),
37                evidence: vec![
38                    format!("QueryName={query}"),
39                    format!("Image={image}"),
40                ],
41                timestamp_ns: ev.timestamp_ns,
42                event_id: ev.event_id,
43                channel: ev.channel.clone(),
44            })
45        })
46        .collect()
47}
48
49fn basename(path: &str) -> &str {
50    path.rsplit(|c| c == '\\' || c == '/').next().unwrap_or(path)
51}
52
53fn is_browser(image: &str) -> bool {
54    let base = basename(image).to_lowercase();
55    BROWSER_PROCESS_NAMES
56        .iter()
57        .any(|b| b.to_lowercase() == base)
58}

60#[cfg(test)]
61mod tests {
62    use super::*;
63    use crate::test_helpers::make_event;
64
65    fn dns_query(image: &str, query: &str) -> EvtxEvent {
66        make_event(
67            EID_SYSMON_DNS_QUERY,
68            SYSMON_CHANNEL,
69            &[("Image", image), ("QueryName", query)],
70        )
71    }
72
73    #[test]
74    fn rundll32_workers_dev_detected() {
75        let ev = dns_query(
76            "C:\\Windows\\System32\\rundll32.exe",
77            "live.itsmartuniverse.workers.dev",
78        );
79        let hits = detect_workers_dev_dns(&[ev]);
80        assert!(!hits.is_empty());
81        assert_eq!(hits[0].kind, EvtxDetectionKind::WorkersDevDnsQuery);
82        assert_eq!(hits[0].mitre_technique_id, "T1102");
83    }
84
85    #[test]
86    fn powershell_workers_dev_detected() {
87        let ev = dns_query(
88            "C:\\Windows\\System32\\WindowsPowerShell\\v1.0\\powershell.exe",
89            "datascience.iotconnectivity.workers.dev",
90        );
91        assert!(!detect_workers_dev_dns(&[ev]).is_empty());
92    }
93
