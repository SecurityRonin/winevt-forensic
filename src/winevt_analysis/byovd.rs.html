1//! Detect BYOVD (Bring Your Own Vulnerable Driver) service installation (T1068).
2
3use forensicnomicon::heuristics::evtx::{
4    BYOVD_DRIVER_NAMES, EID_SERVICE_INSTALLED, EID_SERVICE_INSTALLED_SECURITY,
5};
6use winevt_core::EvtxEvent;
7
8use crate::{EvtxDetection, EvtxDetectionKind};
9
10/// Detect installation of a known-vulnerable driver service.
11///
12/// Matches Security EID 4697 and System EID 7045 (service installed) where
13/// `ServiceName` is in [`BYOVD_DRIVER_NAMES`].  QWCrypt uses the Zemana
14/// AntiMalware driver (`zamguard64`) to terminate EDR/AV at kernel level
15/// before deploying the encryptor (T1068).
16pub fn detect_byovd_driver_install(events: &[EvtxEvent]) -> Vec<EvtxDetection> {
17    events
18        .iter()
19        .filter(|ev| {
20            (ev.event_id == EID_SERVICE_INSTALLED && ev.channel == "System")
21                || (ev.event_id == EID_SERVICE_INSTALLED_SECURITY && ev.channel == "Security")
22        })
23        .filter_map(|ev| {
24            let name = ev.data.get("ServiceName")?;
25            BYOVD_DRIVER_NAMES
26                .iter()
27                .find(|&&pat| name.eq_ignore_ascii_case(pat))
28                .map(|&pat| EvtxDetection {
29                    kind: EvtxDetectionKind::ByovdDriverInstall,
30                    mitre_technique_id: "T1068",
31                    tactic: "Privilege Escalation",
32                    description: format!(
33                        "Known-vulnerable driver service installed: '{name}' — possible BYOVD attack"
34                    ),
35                    evidence: vec![format!("ServiceName={name}"), format!("matched_ioc={pat}")],
36                    timestamp_ns: ev.timestamp_ns,
37                    event_id: ev.event_id,
38                    channel: ev.channel.clone(),
39                })
40        })
41        .collect()
42}

44#[cfg(test)]
45mod tests {
46    use super::*;
47    use crate::test_helpers::make_event;
48
49    #[test]
50    fn zamguard64_service_install_detected() {
51        let ev = make_event(7045, "System", &[("ServiceName", "zamguard64")]);
52        let hits = detect_byovd_driver_install(&[ev]);
53        assert!(!hits.is_empty());
54        assert_eq!(hits[0].kind, EvtxDetectionKind::ByovdDriverInstall);
55        assert_eq!(hits[0].mitre_technique_id, "T1068");
56    }
57
58    #[test]
59    fn zam_variant_detected() {
60        let ev = make_event(7045, "System", &[("ServiceName", "ZAM")]);
61        assert!(!detect_byovd_driver_install(&[ev]).is_empty());
62    }
63
64    #[test]
65    fn security_4697_also_detected() {
66        let ev = make_event(4697, "Security", &[("ServiceName", "ZemanaAntiMalware")]);
67        assert!(!detect_byovd_driver_install(&[ev]).is_empty());
68    }
69
70    #[test]
71    fn other_driver_names_detected() {
72        // RTCore64 is another commonly abused BYOVD driver
73        let ev = make_event(7045, "System", &[("ServiceName", "RTCore64")]);
74        assert!(!detect_byovd_driver_install(&[ev]).is_empty());
75    }
76
77    #[test]
78    fn benign_service_not_detected() {
79        let ev = make_event(7045, "System", &[("ServiceName", "Spooler")]);
80        assert!(detect_byovd_driver_install(&[ev]).is_empty());
81    }
82
83    #[test]
84    fn wrong_event_id_not_detected() {
85        let ev = make_event(4624, "Security", &[("ServiceName", "zamguard64")]);
86        assert!(detect_byovd_driver_install(&[ev]).is_empty());
87    }
88
89    #[test]
90    fn evidence_contains_service_name() {
91        let ev = make_event(7045, "System", &[("ServiceName", "zamguard64")]);
92        let hits = detect_byovd_driver_install(&[ev]);
93        assert!(!hits.is_empty());
94        assert!(hits[0].evidence.iter().any(|e| e.contains("zamguard64")));
95    }
96}
