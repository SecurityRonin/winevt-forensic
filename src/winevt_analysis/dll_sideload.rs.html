1//! Detect srvcli.dll / netutils.dll sideloading via Sysmon EID 7 (T1574.002).
2
3use forensicnomicon::heuristics::evtx::{
4    EID_SYSMON_IMAGE_LOAD, SIDELOAD_HIJACK_DLLS, SYSMON_CHANNEL, SYSTEM_DLL_SAFE_PATH_PREFIXES,
5};
6use winevt_core::EvtxEvent;
7
8use crate::{EvtxDetection, EvtxDetectionKind};
9
10/// Detect DLL sideloading of srvcli.dll or netutils.dll from a non-system path.
11///
12/// Sysmon EID 7 (ImageLoad) fires when a process loads a DLL.  `ImageLoaded`
13/// gives the full path of the loaded DLL.  Both `srvcli.dll` and `netutils.dll`
14/// are legitimate Windows DLLs (loaded by 29+ applications) but are confirmed
15/// as sideloading targets in QWCrypt/RedCurl intrusions — loaded by a renamed
16/// copy of ADNotificationManager.exe from `%APPDATA%`, `%LOCALAPPDATA%\Temp\`,
17/// or `C:\ProgramData\<random>\` (T1574.002).
18pub fn detect_dll_sideload(events: &[EvtxEvent]) -> Vec<EvtxDetection> {
19    events
20        .iter()
21        .filter(|ev| ev.event_id == EID_SYSMON_IMAGE_LOAD && ev.channel == SYSMON_CHANNEL)
22        .filter_map(|ev| {
23            let loaded = ev.data.get("ImageLoaded")?;
24            let base = basename(loaded).to_lowercase();
25            let matched = SIDELOAD_HIJACK_DLLS
26                .iter()
27                .find(|&&dll| dll.to_lowercase() == base)?;
28            if is_safe_system_path(loaded) {
29                return None;
30            }
31            Some(EvtxDetection {
32                kind: EvtxDetectionKind::DllSideloadHijack,
33                mitre_technique_id: "T1574.002",
34                tactic: "Defense Evasion",
35                description: format!(
36                    "DLL sideload: '{matched}' loaded from non-system path '{loaded}'"
37                ),
38                evidence: vec![
39                    format!("ImageLoaded={loaded}"),
40                    format!("matched_dll={matched}"),
41                ],
42                timestamp_ns: ev.timestamp_ns,
43                event_id: ev.event_id,
44                channel: ev.channel.clone(),
45            })
46        })
47        .collect()
48}
49
50fn basename(path: &str) -> &str {
51    path.rsplit(|c| c == '\\' || c == '/').next().unwrap_or(path)
52}
53
54fn is_safe_system_path(path: &str) -> bool {
55    let path_lower = path.to_lowercase();
56    SYSTEM_DLL_SAFE_PATH_PREFIXES
57        .iter()
58        .any(|safe| path_lower.starts_with(&safe.to_lowercase()))
59}

61#[cfg(test)]
62mod tests {
63    use super::*;
64    use crate::test_helpers::make_event;
65
66    fn sideload_event(dll_path: &str) -> EvtxEvent {
67        make_event(
68            EID_SYSMON_IMAGE_LOAD,
69            SYSMON_CHANNEL,
70            &[
71                ("Image", "C:\\Users\\victim\\AppData\\Local\\Temp\\ADNotificationManager.exe"),
72                ("ImageLoaded", dll_path),
73                ("Signed", "false"),
74            ],
75        )
76    }
77
78    #[test]
79    fn srvcli_from_temp_detected() {
80        let ev = sideload_event("C:\\Users\\victim\\AppData\\Local\\Temp\\srvcli.dll");
81        let hits = detect_dll_sideload(&[ev]);
82        assert!(!hits.is_empty());
83        assert_eq!(hits[0].kind, EvtxDetectionKind::DllSideloadHijack);
84        assert_eq!(hits[0].mitre_technique_id, "T1574.002");
85    }
86
87    #[test]
88    fn netutils_from_programdata_detected() {
89        let ev = sideload_event("C:\\ProgramData\\redcurl\\netutils.dll");
90        assert!(!detect_dll_sideload(&[ev]).is_empty());
91    }
92
93    #[test]
94    fn srvcli_from_system32_not_detected() {
95        let ev = sideload_event("C:\\Windows\\System32\\srvcli.dll");
96        assert!(detect_dll_sideload(&[ev]).is_empty());
97    }
98
99    #[test]
100    fn netutils_from_syswow64_not_detected() {
101        let ev = sideload_event("C:\\Windows\\SysWOW64\\netutils.dll");

103    }

129    #[test]
130    fn evidence_contains_dll_path() {
