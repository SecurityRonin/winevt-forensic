1//! Detect Hyper-V VM shutdown events indicating pre-encryption staging (T1486).
2
3use forensicnomicon::heuristics::evtx::{
4    EID_HYPERV_VM_STATE_CHANGE, EID_HYPERV_VM_STOPPED, HYPERV_VMMS_CHANNEL,
5};
6use winevt_core::EvtxEvent;
7
8use crate::{EvtxDetection, EvtxDetectionKind};
9
10/// Detect Hyper-V VM shutdown events in the VMMS Admin channel.
11///
12/// QWCrypt/RedCurl shuts down all Hyper-V VMs before encrypting the VHD/VHDX
13/// files on disk — the guest OS must be stopped for the host to get exclusive
14/// write access.  EID 13002 (state change initiated) and EID 13003 (VM
15/// stopped) in the `Microsoft-Windows-Hyper-V-VMMS/Admin` channel indicate
16/// this pre-encryption staging step (T1486).
17///
18/// Returns one detection per matching event.
19pub fn detect_hyperv_vm_shutdown(events: &[EvtxEvent]) -> Vec<EvtxDetection> {
20    events
21        .iter()
22        .filter(|ev| {
23            ev.channel == HYPERV_VMMS_CHANNEL
24                && (ev.event_id == EID_HYPERV_VM_STATE_CHANGE
25                    || ev.event_id == EID_HYPERV_VM_STOPPED)
26        })
27        .map(|ev| {
28            let vm_name = ev
29                .data
30                .get("VmName")
31                .map(String::as_str)
32                .unwrap_or("<unknown>");
33            EvtxDetection {
34                kind: EvtxDetectionKind::HypervVmShutdown,
35                mitre_technique_id: "T1486",
36                tactic: "Impact",
37                description: format!(
38                    "Hyper-V VM shut down (EID {}): '{vm_name}' — pre-encryption staging",
39                    ev.event_id
40                ),
41                evidence: vec![
42                    format!("VmName={vm_name}"),
43                    format!("event_id={}", ev.event_id),
44                ],
45                timestamp_ns: ev.timestamp_ns,
46                event_id: ev.event_id,
47                channel: ev.channel.clone(),
48            }
49        })
50        .collect()
51}

53#[cfg(test)]
54mod tests {
55    use super::*;
56    use crate::test_helpers::make_event;
57
58    fn hyperv_event(eid: u32) -> winevt_core::EvtxEvent {
59        make_event(eid, HYPERV_VMMS_CHANNEL, &[("VmName", "test-vm-01")])
60    }
61
62    #[test]
63    fn eid_13002_vm_state_change_detected() {
64        let ev = hyperv_event(EID_HYPERV_VM_STATE_CHANGE);
65        let hits = detect_hyperv_vm_shutdown(&[ev]);
66        assert!(!hits.is_empty());
67        assert_eq!(hits[0].kind, EvtxDetectionKind::HypervVmShutdown);
68        assert_eq!(hits[0].mitre_technique_id, "T1486");
69    }
70
71    #[test]
72    fn eid_13003_vm_stopped_detected() {
73        let ev = hyperv_event(EID_HYPERV_VM_STOPPED);
74        assert!(!detect_hyperv_vm_shutdown(&[ev]).is_empty());
75    }
76
77    #[test]
78    fn wrong_channel_not_detected() {
79        // Same EID in a different channel is not a Hyper-V event
80        let ev = make_event(EID_HYPERV_VM_STATE_CHANGE, "System", &[]);
81        assert!(detect_hyperv_vm_shutdown(&[ev]).is_empty());
82    }
83
84    #[test]
85    fn unrelated_hyperv_event_not_detected() {
86        let ev = make_event(9999, HYPERV_VMMS_CHANNEL, &[]);
87        assert!(detect_hyperv_vm_shutdown(&[ev]).is_empty());
88    }
89
90    #[test]
91    fn evidence_mentions_vm_name() {
92        let ev = hyperv_event(EID_HYPERV_VM_STOPPED);
93        let hits = detect_hyperv_vm_shutdown(&[ev]);
94        assert!(!hits.is_empty());
95        assert!(
96            hits[0].evidence.iter().any(|e| e.contains("test-vm-01"))
97                || hits[0].description.contains("test-vm-01")
98        );
99    }
100
101    #[test]
102    fn cluster_of_shutdowns_produces_multiple_detections() {
103        let events: Vec<_> = (0..3)
104            .map(|i| make_event(
105                EID_HYPERV_VM_STOPPED,
106                HYPERV_VMMS_CHANNEL,
107                &[("VmName", &format!("vm-{i:02}"))],
108            ))
109            .collect();
110        assert_eq!(detect_hyperv_vm_shutdown(&events).len(), 3);
111    }
112}
