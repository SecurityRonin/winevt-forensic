1//! Detect PowerShell history file deletion via Sysmon EID 23/26 (T1070.003).
2
3use forensicnomicon::heuristics::evtx::{
4    EID_SYSMON_FILE_DELETE, EID_SYSMON_FILE_DELETE_DETECTED, PS_HISTORY_PATH_FRAGMENT,
5    SYSMON_CHANNEL,
6};
7use winevt_core::EvtxEvent;
8
9use crate::{EvtxDetection, EvtxDetectionKind};
10
11/// Detect deletion of `ConsoleHost_history.txt` — the PowerShell PSReadLine
12/// command history file.
13///
14/// QWCrypt/RedCurl's cleanup batch script deletes PowerShell history across all
15/// user profiles (T1070.003).  Sysmon EID 23 (FileDelete, requires archiving)
16/// or EID 26 (FileDeleteDetected, v13+) fires when `TargetFilename` matches the
17/// path fragment `ConsoleHost_history.txt`.  A single deletion may be a user
18/// action; deletions across multiple user profiles within seconds indicate
19/// automated ransomware cleanup.
20pub fn detect_ps_history_wipe(events: &[EvtxEvent]) -> Vec<EvtxDetection> {
21    events
22        .iter()
23        .filter(|ev| {
24            (ev.event_id == EID_SYSMON_FILE_DELETE || ev.event_id == EID_SYSMON_FILE_DELETE_DETECTED)
25                && ev.channel == SYSMON_CHANNEL
26        })
27        .filter_map(|ev| {
28            let path = ev.data.get("TargetFilename")?;
29            if !path.contains(PS_HISTORY_PATH_FRAGMENT) {
30                return None;
31            }
32            Some(EvtxDetection {
33                kind: EvtxDetectionKind::PsHistoryWipe,
34                mitre_technique_id: "T1070.003",
35                tactic: "Defense Evasion",
36                description: format!("PowerShell history file deleted: '{path}'"),
37                evidence: vec![format!("TargetFilename={path}")],
38                timestamp_ns: ev.timestamp_ns,
39                event_id: ev.event_id,
40                channel: ev.channel.clone(),
41            })
42        })
43        .collect()
44}

46#[cfg(test)]
47mod tests {
48    use super::*;
49    use crate::test_helpers::make_event;
50
51    fn history_delete(eid: u32, path: &str) -> EvtxEvent {
52        make_event(eid, SYSMON_CHANNEL, &[("TargetFilename", path)])
53    }
54
55    const HISTORY_PATH: &str =
56        "C:\\Users\\victim\\AppData\\Roaming\\Microsoft\\Windows\\PowerShell\\PSReadLine\\ConsoleHost_history.txt";
57
58    #[test]
59    fn eid23_history_wipe_detected() {
60        let ev = history_delete(EID_SYSMON_FILE_DELETE, HISTORY_PATH);
61        let hits = detect_ps_history_wipe(&[ev]);
62        assert!(!hits.is_empty());
63        assert_eq!(hits[0].kind, EvtxDetectionKind::PsHistoryWipe);
64        assert_eq!(hits[0].mitre_technique_id, "T1070.003");
65    }
66
67    #[test]
68    fn eid26_history_wipe_detected() {
69        let ev = history_delete(EID_SYSMON_FILE_DELETE_DETECTED, HISTORY_PATH);
70        let hits = detect_ps_history_wipe(&[ev]);
71        assert!(!hits.is_empty());
72        assert_eq!(hits[0].kind, EvtxDetectionKind::PsHistoryWipe);
73    }
74
75    #[test]
76    fn benign_file_delete_not_detected() {
77        let ev = history_delete(EID_SYSMON_FILE_DELETE, "C:\\Users\\victim\\Desktop\\report.docx");
78        assert!(detect_ps_history_wipe(&[ev]).is_empty());
79    }
80
81    #[test]
82    fn wrong_channel_not_detected() {
83        let ev = make_event(
84            EID_SYSMON_FILE_DELETE,
85            "Security",
86            &[("TargetFilename", HISTORY_PATH)],
87        );
88        assert!(detect_ps_history_wipe(&[ev]).is_empty());
89    }
90
91    #[test]
92    fn wrong_event_id_not_detected() {
93        let ev = make_event(11, SYSMON_CHANNEL, &[("TargetFilename", HISTORY_PATH)]);
94        assert!(detect_ps_history_wipe(&[ev]).is_empty());
95    }
96
97    #[test]
98    fn multiple_user_wipes_produce_multiple_detections() {
99        let events = vec![
100            history_delete(
101                EID_SYSMON_FILE_DELETE,
102                "C:\\Users\\alice\\AppData\\Roaming\\Microsoft\\Windows\\PowerShell\\PSReadLine\\ConsoleHost_history.txt",
103            ),
104            history_delete(
105                EID_SYSMON_FILE_DELETE,
106                "C:\\Users\\bob\\AppData\\Roaming\\Microsoft\\Windows\\PowerShell\\PSReadLine\\ConsoleHost_history.txt",
107            ),
108        ];
109        assert_eq!(detect_ps_history_wipe(&events).len(), 2);
110    }
111
112    #[test]
113    fn evidence_contains_file_path() {
114        let ev = history_delete(EID_SYSMON_FILE_DELETE, HISTORY_PATH);
115        let hits = detect_ps_history_wipe(&[ev]);
116        assert!(!hits.is_empty());
117        let combined = hits[0].evidence.join(" ");
118        assert!(combined.contains("ConsoleHost_history.txt"));
119    }
