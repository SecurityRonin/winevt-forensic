1//! Detect shadow copy / VSS deletion events (T1490).
2
3use forensicnomicon::heuristics::evtx::{EID_VSS_ERROR, EID_VSS_SNAPSHOT_DELETED};
4use winevt_core::EvtxEvent;
5
6use crate::{EvtxDetection, EvtxDetectionKind};
7
8/// Detect VSS shadow copy deletion events.
9///
10/// Fires on Application channel EID 8193 (VSS service error — fired when
11/// `vssadmin delete shadows` or `wmic shadowcopy delete` runs) and EID 524
12/// (snapshot deleted).  QWCrypt deletes shadow copies to prevent recovery
13/// of encrypted Hyper-V VHD/VHDX files (T1490).
14///
15/// Returns one detection per matching event.
16pub fn detect_vss_deletion(events: &[EvtxEvent]) -> Vec<EvtxDetection> {
17    events
18        .iter()
19        .filter(|ev| {
20            ev.channel == "Application"
21                && (ev.event_id == EID_VSS_ERROR || ev.event_id == EID_VSS_SNAPSHOT_DELETED)
22        })
23        .map(|ev| EvtxDetection {
24            kind: EvtxDetectionKind::VssDeletion,
25            mitre_technique_id: "T1490",
26            tactic: "Impact",
27            description: format!(
28                "VSS shadow copy deletion event (EID {}): inhibit system recovery",
29                ev.event_id
30            ),
31            evidence: vec![format!("event_id={}", ev.event_id), format!("channel={}", ev.channel)],
32            timestamp_ns: ev.timestamp_ns,
33            event_id: ev.event_id,
34            channel: ev.channel.clone(),
35        })
36        .collect()
37}

39#[cfg(test)]
40mod tests {
41    use super::*;
42    use crate::test_helpers::make_event;
43
44    #[test]
45    fn eid_8193_vss_error_detected() {
46        let ev = make_event(EID_VSS_ERROR, "Application", &[("Source", "VSS")]);
47        let hits = detect_vss_deletion(&[ev]);
48        assert!(!hits.is_empty());
49        assert_eq!(hits[0].kind, EvtxDetectionKind::VssDeletion);
50        assert_eq!(hits[0].mitre_technique_id, "T1490");
51    }
52
53    #[test]
54    fn eid_524_snapshot_deleted_detected() {
55        let ev = make_event(EID_VSS_SNAPSHOT_DELETED, "Application", &[]);
56        assert!(!detect_vss_deletion(&[ev]).is_empty());
57    }
58
59    #[test]
60    fn unrelated_application_event_not_detected() {
61        let ev = make_event(1000, "Application", &[]);
62        assert!(detect_vss_deletion(&[ev]).is_empty());
63    }
64
65    #[test]
66    fn security_8193_wrong_channel_not_detected() {
67        // EID 8193 in Security channel is unrelated
68        let ev = make_event(EID_VSS_ERROR, "Security", &[]);
69        assert!(
70            detect_vss_deletion(&[ev]).is_empty(),
71            "EID 8193 in Security channel is not a VSS event"
72        );
73    }
74
75    #[test]
76    fn multiple_vss_events_produce_multiple_detections() {
77        let events = vec![
78            make_event(EID_VSS_ERROR, "Application", &[]),
79            make_event(EID_VSS_SNAPSHOT_DELETED, "Application", &[]),
80        ];
81        assert_eq!(detect_vss_deletion(&events).len(), 2);
82    }
