1//! One-click E01 → EVTX extraction + Hayabusa report pipeline.
2//!
3//! # Pipeline
4//!
5//! 1. Open the E01 via [`ewf::EwfReader`] (`Read + Seek`).
6//! 2. Parse the MBR to locate the NTFS partition offset.
7//! 3. Traverse NTFS with the [`ntfs`] crate to find `*.evtx` files under
8//!    `Windows/System32/winevt/Logs/`.
9//! 4. Stream each EVTX file out to a caller-supplied output directory.
10//! 5. (CLI layer) Run Hayabusa on that directory and emit a combined JSON report.
11
12use std::path::{Path, PathBuf};
13
14pub use error::TriageError;
15
16mod error;
17mod extract;
18mod mbr;
19mod partition;

21// ── Public types ─────────────────────────────────────────────────────────────
22
23/// A single EVTX file extracted from the disk image.
24#[derive(Debug, serde::Serialize)]
25pub struct ExtractedEvtx {
26    /// Base filename, e.g. `"Security.evtx"`.
27    pub name: String,
28    /// Absolute path to the extracted file in `out_dir`.
29    pub path: PathBuf,
30    /// File size in bytes.
31    pub size: u64,
32}
33
34/// Result of the EVTX extraction phase.
35#[derive(Debug, serde::Serialize)]
36pub struct TriageReport {
37    /// Canonical path to the source E01 image.
38    pub image: PathBuf,
39    /// NTFS partition start in 512-byte sectors (from MBR).
40    pub ntfs_offset_sectors: u64,
41    /// Extracted EVTX files.
42    pub evtx_files: Vec<ExtractedEvtx>,
43}
44
45// ── Public API ───────────────────────────────────────────────────────────────
46
47/// Parse MBR sector 0 (512 bytes) and return the LBA start of the first
48/// NTFS partition (type `0x07`, `0x17`, or `0x27`).
49///
50/// Returns `None` when no NTFS partition entry is found.
51pub fn parse_mbr_ntfs_offset(sector: &[u8; 512]) -> Option<u64> {
52    mbr::parse_ntfs_offset(sector)
53}
54
55/// Extract all `*.evtx` files from an E01 forensic image into `out_dir`.
56///
57/// Opens the image via [`ewf::EwfReader`], locates the NTFS partition via the
58/// MBR, traverses `Windows/System32/winevt/Logs/`, and writes each file to
59/// `out_dir/<filename>.evtx`.
60pub fn extract_evtx_from_e01(e01_path: &Path, out_dir: &Path) -> Result<TriageReport, TriageError> {
61    extract::run(e01_path, out_dir)
62}
