1//! MBR partition table parser — pure, no I/O.
2
3/// Partition table starts at offset 0x1BE; 4 entries × 16 bytes each.
4/// Entry layout: status(1) + chs_first(3) + type(1) + chs_last(3) + lba_start(4) + sectors(4).
5const TABLE_OFFSET: usize = 0x1BE;
6const ENTRY_SIZE: usize = 16;
7const TYPE_OFFSET: usize = 4;
8const LBA_OFFSET: usize = 8;
9
10const NTFS_TYPES: &[u8] = &[0x07, 0x17, 0x27];
11
12/// Return the LBA start sector of the first NTFS partition, or `None`.
13pub(crate) fn parse_ntfs_offset(sector: &[u8; 512]) -> Option<u64> {
14    for i in 0..4usize {
15        let base = TABLE_OFFSET + i * ENTRY_SIZE;
16        let part_type = sector[base + TYPE_OFFSET];
17        if NTFS_TYPES.contains(&part_type) {
18            let lba = u32::from_le_bytes([
19                sector[base + LBA_OFFSET],
20                sector[base + LBA_OFFSET + 1],
21                sector[base + LBA_OFFSET + 2],
22                sector[base + LBA_OFFSET + 3],
23            ]);
24            return Some(u64::from(lba));
25        }
26    }
27    None
28}
